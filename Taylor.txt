-- ============================================================
-- attach gui
-- ============================================================
local function attach_gui(gui)
    if not gui then return nil end
    local cands = {}
    if type(gethui) == "function" then
        local ok, h = pcall(gethui)
        if ok and h then cands[#cands+1] = h end
    end
    cands[#cands+1] = CoreGuiSvc
    local pg = LP:FindFirstChildOfClass("PlayerGui")
    if pg then cands[#cands+1] = pg end
    for i = 1, #cands do
        local ok = pcall(function() gui.Parent = cands[i] end)
        if ok and gui.Parent then return cands[i] end
    end
end

local function notify(text, dur)
    dur = dur or 3
    local gui = Instance.new("ScreenGui")
    gui.Name="\0"; gui.ResetOnSpawn=false; gui.IgnoreGuiInset=true; gui.DisplayOrder=1000
    if not attach_gui(gui) then return end

    local f = Instance.new("Frame")
    f.BackgroundColor3 = T.card
    f.BorderSizePixel = 0
    f.Size = UDim2.fromOffset(280, 50)
    f.Position = UDim2.new(1, -300, 0, 60)
    f.Parent = gui
    local uc = Instance.new("UICorner"); uc.CornerRadius = UDim.new(0, 8); uc.Parent = f

    local auc = Instance.new("UICorner"); auc.CornerRadius = UDim.new(1, 0); auc.Parent = acc

    local t = Instance.new("TextLabel")
    t.BackgroundTransparency = 1
    t.Size = UDim2.new(1, -24, 1, 0)
    t.Position = UDim2.fromOffset(14, 0)
    t.Font = Enum.Font.Gotham
    t.Text = text
    t.TextColor3 = T.text
    t.TextSize = 13
    t.TextXAlignment = Enum.TextXAlignment.Left
    t.TextWrapped = true
    t.Parent = f

    pcall(function()
        TweenService:Create(f, TweenInfo.new(0.22, Enum.EasingStyle.Quart), {Position = UDim2.new(1, -300, 0, 72)}):Play()
    end)
    task.delay(dur, function()
        pcall(function()
            TweenService:Create(f, TweenInfo.new(0.28), {BackgroundTransparency = 1, Position = UDim2.new(1, -280, 0, 72)}):Play()
            TweenService:Create(t, TweenInfo.new(0.28), {TextTransparency = 1}):Play()

            TweenService:Create(acc, TweenInfo.new(0.28), {BackgroundTransparency = 1}):Play()
        end)
        task.wait(0.35)
        pcall(function() gui:Destroy() end)
    end)

function UI.new(cfg)
    cfg = cfg or {}
    local self = setmetatable({}, UI)
    self.title = cfg.title or "SHITARO"
    self.enabled = true
    self.key = Enum.KeyCode.Insert
    self.tabs = {}
    self.tab = nil
    self.mobile = UserInputService.TouchEnabled and true or false

    local vp = CAM and CAM.ViewportSize or Vector2.new(1280, 720)
    self.size = Vector2.new(math.min(vp.X - 40, 780), math.min(vp.Y - 80, 460))
    self.pos = Vector2.new((vp.X - self.size.X) / 2, math.max(20, (vp.Y - self.size.Y) / 2 - 40))

    self.ok, self.err = pcall(function() self:_build() end)
    if not self.ok then warn("[SHITARO UI]", self.err) end
    return self
end

function UI:toggle(state)
    if not self.win then return end
    self.enabled = state and true or not self.enabled
    self.win.Visible = self.enabled
end

-- TAB (sidebar icon + label)
function UI:tab(cfg)
    cfg = cfg or {}
    if not self.ok then return nil end
    local name = cfg.name or "tab"
    local icon = cfg.icon or TAB_ICONS[name] or "●"

    local tab = {name=name, icon=icon, btn=nil, page=nil}

    local b = Instance.new("TextButton")
    b.BackgroundColor3 = T.sidebar
    b.BackgroundTransparency = 1
    b.BorderSizePixel = 0
    b.Size = UDim2.new(0, 62, 0, 48)
    b.Text = ""
    b.AutoButtonColor = false
    b.Parent = self.side
    local buc = Instance.new("UICorner"); buc.CornerRadius = UDim.new(0, 7); buc.Parent = b

    local ic = Instance.new("TextLabel")
    ic.BackgroundTransparency = 1
    ic.Size = UDim2.new(1, 0, 0, 26)
    ic.Position = UDim2.fromOffset(0, 4)
    ic.Font = Enum.Font.GothamBold
    ic.Text = icon
    ic.TextColor3 = T.muted
    ic.TextSize = 18
    ic.Parent = b

    local lb = Instance.new("TextLabel")
    lb.BackgroundTransparency = 1
    lb.Size = UDim2.new(1, 0, 0, 12)
    lb.Position = UDim2.new(0, 0, 1, -16)
    lb.Font = Enum.Font.Gotham
    lb.Text = name
    lb.TextColor3 = T.muted
    lb.TextSize = 9
    lb.Parent = b

    tab.btn = b
    tab.ic = ic
    tab.lb = lb

    self.tabs[#self.tabs+1] = tab
    tab.page = self:_page()
    tab.page.Visible = false

    b.MouseButton1Click:Connect(function() self:selectTab(tab) end)
    b.MouseEnter:Connect(function() if self.tab ~= tab then b.BackgroundTransparency = 0; b.BackgroundColor3 = T.row end end)
    b.MouseLeave:Connect(function() if self.tab ~= tab then b.BackgroundTransparency = 1 end end)

    if #self.tabs == 1 then self:selectTab(tab) end
    return tab
end

function UI:selectTab(tab)
    if not tab or not tab.page then return end
    self.tab = tab
    for _, t in ipairs(self.tabs) do
        if t == tab then
            t.btn.BackgroundTransparency = 0
            t.btn.BackgroundColor3 = T.row
            t.ic.TextColor3 = T.accent
            t.lb.TextColor3 = T.accent
            t.page.Visible = true
        else
            t.btn.BackgroundTransparency = 1
            t.ic.TextColor3 = T.muted
            t.lb.TextColor3 = T.muted
            t.page.Visible = false
        end
    end
end

function UI:_page()
    local p = Instance.new("ScrollingFrame")
    p.BackgroundTransparency = 1
    p.BorderSizePixel = 0
    p.Size = UDim2.fromScale(1, 1)
    p.ScrollBarThickness = self.mobile and 6 or 4
    p.ScrollBarImageColor3 = T.border2
    p.ScrollBarImageTransparency = 0.4
    p.CanvasSize = UDim2.new(0, 0, 0, 0)
    p.AutomaticCanvasSize = Enum.AutomaticSize.Y
    p.ScrollingDirection = Enum.ScrollingDirection.Y
    p.Parent = self.body

    -- 3 columns: left, center, right (as shitaro)
    local pad = 10
    local gap = 8
    local colw = (1 / 3)

    local left = Instance.new("Frame")
    left.BackgroundTransparency = 1
    left.BorderSizePixel = 0
    left.Size = UDim2.new(colw, -pad - gap/1.5, 1, -pad*2)
    left.Position = UDim2.fromOffset(pad, pad)
    left.Parent = p
    local lll = Instance.new("UIListLayout"); lll.Padding = UDim.new(0, gap); lll.SortOrder = Enum.SortOrder.LayoutOrder; lll.Parent = left
    local leftAuto = Instance.new("Frame")
    leftAuto.BackgroundTransparency = 1
    leftAuto.Size = UDim2.new(1, 0, 0, 0)
    leftAuto.AutomaticSize = Enum.AutomaticSize.Y
    leftAuto.Parent = left

    -- Actually we need the columns inside a scrolling parent. Use 3 ScrollingFrames side by side.
    left:Destroy()
    leftAuto:Destroy()

    local function mcol(px)
        local c = Instance.new("ScrollingFrame")
        c.BackgroundTransparency = 1
        c.BorderSizePixel = 0
        c.Size = UDim2.new(colw, -pad - gap*2/3, 1, -pad*2)
        c.Position = UDim2.fromOffset(px, pad)
        c.ScrollBarThickness = 3
        c.ScrollBarImageColor3 = T.border2
        c.ScrollBarImageTransparency = 0.5
        c.CanvasSize = UDim2.new(0, 0, 0, 0)
        c.AutomaticCanvasSize = Enum.AutomaticSize.Y
        c.ScrollingDirection = Enum.ScrollingDirection.Y
        c.Parent = p
        local ll = Instance.new("UIListLayout"); ll.Padding = UDim.new(0, gap); ll.SortOrder = Enum.SortOrder.LayoutOrder; ll.Parent = c
        return c, ll
    end

    local colL, llL = mcol(pad)
    local colC, llC = mcol(pad + colw * self.size.X / 3 + gap/2)
    local colR, llR = mcol(pad + colw * self.size.X * 2 / 3 + gap)

    -- recalibrate width/position properly (we're inside a Frame of body, so use Scale from parent frame)
    local function fix(c, scale, off)
        c.Size = UDim2.new(scale, -pad - gap*2/3, 1, -pad*2)
        c.Position = UDim2.fromOffset(off, pad)
    end

    -- Actually simpler: 3 columns of equal width with gap
    local totalW = self.size.X - 76 -- sidebar removed
    -- We just use Scale thirds; gap handled via size offset:
    for i, c in ipairs({colL, colC, colR}) do
        c.Size = UDim2.new(1/3, -pad*2/3 - gap*2/3, 1, -pad*2)
        c.Position = UDim2.new((i-1)/3, pad + ((i-1)*gap)/3, 0, pad)
    end
    colC.Position = UDim2.new(1/3, pad + gap/3, 0, pad)
    colR.Position = UDim2.new(2/3, pad + gap*2/3, 0, pad)

    p._cols = {left=colL, center=colC, right=colR}
    p._lls = {left=llL, center=llC, right=llR}

    return p
end

local BT = {hist={}, keep=2.0}
local function bt_track()
    local now = os.clock()
    local co = now - BT.keep
    for _, plr in ipairs(Players:GetPlayers()) do
        if plr == LP then continue end
        local c = plr.Character
        local hrp = c and c:FindFirstChild("HumanoidRootPart")
        local hu = c and c:FindFirstChildOfClass("Humanoid")
        if hrp and hu and hu.Health > 0 then
            local h = BT.hist[plr.Name]
            if not h then h = {}; BT.hist[plr.Name] = h end
            h[#h+1] = {t=now, pos=hrp.Position}
            local i = 1
            while i <= #h and h[i].t < co do i = i + 1 end
            if i > 1 then
                for j = 1, #h - (i - 1) do h[j] = h[j + i - 1] end
                for j = #h - (i - 1) + 1, #h do h[j] = nil end
            end
        end
    end
    for n in pairs(BT.hist) do
        if not Players:FindFirstChild(n) then BT.hist[n] = nil end
    end
end

local function bt_get(name, ms)
    local h = BT.hist[name]
    if not h or #h == 0 then return nil end
    local tgt = os.clock() - (ms / 1000)
    local best, bd = h[1], math.huge
    for i = 1, #h do
        local d = math.abs(h[i].t - tgt)
        if d < bd then best, bd = h[i], d end
    end
    return best and best.pos
end

local TGT = {player=nil, char=nil, part=nil, hum=nil}

local function refresh_target()
    local found = nil
    if C.priority == "murderer" then
        local m = get_round()
        local d = m and m.PlayerData
        if type(d) == "table" then
            for name, info in pairs(d) do
                if type(info) == "table" and info.Role == "Murderer" and not info.Dead then
                    found = Players:FindFirstChild(name); if found then break end
                end
            end
        end
        if not found then
            for _, plr in ipairs(Players:GetPlayers()) do
                if plr ~= LP and holds(plr.Character, "Knife") then found = plr; break end
            end
        end
    end
    if not found then
        local myhrp = LP.Character and LP.Character:FindFirstChild("HumanoidRootPart")
        if myhrp then
            local best, bd = nil, math.huge
            for _, plr in ipairs(Players:GetPlayers()) do
                if plr ~= LP and plr.Character then
                    local h = plr.Character:FindFirstChild("HumanoidRootPart")
                    local hu = plr.Character:FindFirstChildOfClass("Humanoid")
                    if h and hu and hu.Health > 0 then
                        local d = (h.Position - myhrp.Position).Magnitude
                        if d < bd and d <= C.range then best, bd = plr, d end
                    end
                end
            end
            found = best
        end
    end
    if found ~= TGT.player then TGT.player = found; TGT.char = nil; TGT.part = nil; TGT.hum = nil end
    if not found then return end
    local char = found.Character
    if char ~= TGT.char then TGT.char = char; TGT.part = nil; TGT.hum = nil end
    if not char then return end
    if not TGT.part or not TGT.part.Parent then
        TGT.part = char:FindFirstChild("HumanoidRootPart") or char:FindFirstChild("UpperTorso") or char:FindFirstChild("Torso")
    end
    if not TGT.hum or not TGT.hum.Parent then
        TGT.hum = char:FindFirstChildOfClass("Humanoid")
    end
end

-- ============================================================
-- MOTION
-- ============================================================
local SN = {n=48, i=0, cnt=0, t={}, p={}, last=nil, lt=0, vel=Vector3.zero}
for i = 1, SN.n do SN.t[i] = 0; SN.p[i] = Vector3.zero end

local function sn_push(now, pos)
    SN.i = SN.i % SN.n + 1
    SN.t[SN.i] = now; SN.p[SN.i] = pos
    if SN.cnt < SN.n then SN.cnt = SN.cnt + 1 end
end
local function sn_get(k)
    local idx = (SN.i - k - 1) % SN.n + 1
    return SN.t[idx], SN.p[idx]
end
local function fit_vel()
    if SN.cnt < 3 then return nil end
    local newest = sn_get(0)
    local used, sd = 0, 0
    for k = 0, SN.cnt - 1 do
        local t = sn_get(k)
        if newest - t > 0.4 then break end
        used = used + 1; sd = sd + (t - newest)
    end
    if used < 3 then return nil end
    local md = sd / used
    local num, den = Vector3.zero, 0
    for k = 0, used - 1 do
        local t, p = sn_get(k)
        local d = (t - newest) - md
        num = num + p * d; den = den + d * d
    end
    if den < 1e-8 then return nil end
    return num / den
end
local function recent_vel()
    if SN.cnt < 2 then return nil end
    local newest, head = sn_get(0)
    local best = nil
    for k = 1, SN.cnt - 1 do
        local t, p = sn_get(k)
        local dt = newest - t
        if dt > 0.35 then break end
        best = (head - p) / dt
    end
    return best
end
local function track_motion(now)
    if not TGT.part or not TGT.part.Parent then SN.cnt=0; SN.i=0; SN.last=nil; return end
    local pos = TGT.part.Position
    if not SN.last then SN.last = pos; SN.lt = now; sn_push(now, pos); return end
    if now - SN.lt < 0.005 then return end
    SN.lt = now; SN.last = pos
    sn_push(now, pos)
    local fit = fit_vel()
    local fast = recent_vel()
    if fit and fast then
        local sf = Vector3.new(fit.X, 0, fit.Z).Magnitude
        local ss = Vector3.new(fast.X, 0, fast.Z).Magnitude
        if sf > 200 or sf ~= sf then fit = nil end
        if ss > 200 or ss ~= ss then fast = nil end
    end
    if fit and fast then
        SN.vel = Vector3.new(fast.X*0.7 + fit.X*0.3, fast.Y, fast.Z*0.7 + fit.Z*0.3)
    elseif fast then SN.vel = fast
    elseif fit then SN.vel = fit end
    local h = Vector3.new(SN.vel.X, 0, SN.vel.Z)
    if h.Magnitude > 120 then SN.vel = Vector3.new(SN.vel.X*0.4, SN.vel.Y, SN.vel.Z*0.4) end
end
local function clear_motion()
    SN.cnt=0; SN.i=0; SN.last=nil; SN.lt=0; SN.vel=Vector3.zero
end

local HIT = {"HumanoidRootPart","UpperTorso","Torso","LowerTorso","Head",
    "RightUpperArm","LeftUpperArm","RightUpperLeg","LeftUpperLeg"}

local function aim_point()
    if not TGT.char or not TGT.char.Parent then return nil end
    if not TGT.hum or TGT.hum.Health <= 0 then return nil end
    if C.backtrack then
        local b = bt_get(TGT.player and TGT.player.Name or "", C.bt_win)
        if b then return b end
    end
    local part = TGT.part
    if not part then
        for _, n in ipairs(HIT) do local p = TGT.char:FindFirstChild(n); if p then part = p; break end end
    end
    if not part then return nil end
    local base = part.Position
    if not C.predict then return base end
    if not PING.seen then return base end
    local t = math.clamp(PING.rtt, 0, 0.5)
    if t <= 0 then return base end
    local v = SN.vel
    local x = base.X + v.X * t
    local z = base.Z + v.Z * t
    local y = base.Y
    if math.abs(v.Y) > 1 then y = base.Y + v.Y * t - 0.5 * workspace.Gravity * t * t end
    return Vector3.new(x, y, z)
end

    if not att then return false end
    if force_att and force_att ~= att then restore_force() end
    if not force_att then
        local ok, saved = pcall(function() return att.CFrame end)
        if not ok or typeof(saved) ~= "CFrame" then return false end
        force_att = att; force_orig = saved
    end
    force_stamp = os.clock()
    local ok = pcall(function() att.WorldCFrame = cf end)
    if not ok then restore_force(); return false end
    task.defer(restore_force)
    return true
end
local function origin_cf()
    local c = LP.Character
    local hrp = c and c:FindFirstChild("HumanoidRootPart")
    if not hrp then return nil end
    local att = hrp:FindFirstChild("GunRaycastAttachment")
    return att and att.WorldCFrame or hrp.CFrame
end
local function resolve_force()
    if not TGT.part or not TGT.part.Parent then return nil, nil end
    local live = TGT.part.Position
    local v = SN.vel
    local t = math.clamp(PING.rtt, 0, 0.5)
    local aim = C.predict and Vector3.new(
        live.X + v.X*t,
        live.Y + (math.abs(v.Y) > 1 and (v.Y*t - 0.5*workspace.Gravity*t*t) or 0),
        live.Z + v.Z*t
    ) or live
    local back = live - (aim - live).Unit * C.stand_off
    return CFrame.new(back, aim), CFrame.new(aim)
end

-- auto shoot
local function get_gun()
    local c = LP.Character
    if c then local g = c:FindFirstChild("Gun"); if g then return g, true end end
    local bp = LP:FindFirstChildOfClass("Backpack")
    if bp then local g = bp:FindFirstChild("Gun"); if g then return g, false end end
end
local last_shot = 0
local function auto_fire(now)
    if not C.auto or not C.aim or not tgt_alive() then return end
    local role = my_role()
    if role ~= "Sheriff" and role ~= "Hero" and not get_gun() then return end
    local gun, eq = get_gun()
    if not gun then return end
    if not eq then
        local hum = LP.Character and LP.Character:FindFirstChildOfClass("Humanoid")
        if hum then pcall(function() hum:EquipTool(gun) end) end
        return
    end
    local delay = math.max(C.auto_delay, 0.05)
    if now - last_shot < delay then return end
    local p = aim_point()
    if not p then return end
    local remote = gun:FindFirstChild("Shoot")
    if not remote then
        for _, d in ipairs(gun:GetDescendants()) do
            if d:IsA("RemoteEvent") then remote = d; break end
        end
    end
    if not remote or not remote:IsA("RemoteEvent") then return end
    if C.force then
        local oc, ac = resolve_force()
        if oc and ac then
            if push_origin(oc) then
                pcall(function() remote:FireServer(oc, ac) end)
                last_shot = now
                return
            end
        end
    end
    local oc = origin_cf()
    if oc then
        pcall(function() remote:FireServer(oc, CFrame.new(p)) end)
        last_shot = now
    end
end

    pb.Transparency=1; pb.Anchored=true; pb.CanCollide=false; pb.CanQuery=false
    pb.Size=Vector3.new(1,1,1); pb.CFrame=CFrame.new(b)

    beam.Width0=0.25; beam.Width1=0.25
    beam.LightEmission=3; beam.LightInfluence=0; beam.Brightness=2.5
    beam.Texture="rbxassetid://12781800668"
    beam.Color=ColorSequence.new(C.tracer_col)

    beam.Attachment0=pa.Attachment; beam.Attachment1=pb.Attachment

        pcall(function()
            if beam.Parent then
                TweenService:Create(beam, TweenInfo.new(0.2, Enum.EasingStyle.Linear), {Width0=0, Width1=0}):Play()

            local pa, pb = to_pos(a), to_pos(b)
            if pa and pb then task.spawn(make_tracer, pa, pb) end
        end
    end)
end

local function esp_update()
    if not HAS_DR then return end
    if not C.esp then
        for _, e in pairs(esp_cache) do
            e.box.Visible=false; e.name.Visible=false; e.dist.Visible=false; e.hp.Visible=false
            for _, l in ipairs(e.skel) do l.Visible = false end
        end
        return
    end
    for _, plr in ipairs(Players:GetPlayers()) do
        if plr == LP then continue end
        local char = plr.Character
        local hrp = char and char:FindFirstChild("HumanoidRootPart")
        local hum = char and char:FindFirstChildOfClass("Humanoid")
        local e = esp_cache[plr]
        if char and hrp and hum and hum.Health > 0 then
            if not e then e = new_esp(plr) end
            if e then
                local top = CAM:WorldToViewportPoint(hrp.Position + Vector3.new(0, 3, 0))
                local bot = CAM:WorldToViewportPoint(hrp.Position - Vector3.new(0, 3, 0))
                if top.Z > 0 and bot.Z > 0 then
                    local h = math.abs(top.Y - bot.Y)
                    local w = h * 0.55
                    local bx = bot.X - w / 2
                    local by = top.Y
                    if C.esp_box then
                        e.box.Color = role_color(plr)
                        e.box.Position = Vector2.new(bx, by)
                        e.box.Size = Vector2.new(w, h)
                        e.box.Visible = true
                    else e.box.Visible = false end
                    if C.esp_name then
                        e.name.Color = C.esp_name_col
                        e.name.Text = plr.DisplayName
                        e.name.Position = Vector2.new(bot.X, by - 16)
                        e.name.Visible = true
                    else e.name.Visible = false end
                    if C.esp_dist then
                        local myhrp = LP.Character and LP.Character:FindFirstChild("HumanoidRootPart")
                        local d = myhrp and (hrp.Position - myhrp.Position).Magnitude or 0
                        e.dist.Color = C.esp_dist_col
                        e.dist.Text = string.format("[%d]", math.floor(d + 0.5))
                        e.dist.Position = Vector2.new(bot.X, bot.Y + 2)
                        e.dist.Visible = true
                    else e.dist.Visible = false end
                    if C.esp_hp then
                        local frac = math.clamp(hum.Health / math.max(hum.MaxHealth, 1), 0, 1)
                        e.hp.Color = C.esp_hp_col
                        e.hp.Position = Vector2.new(bx - 5, by + h * (1 - frac))
                        e.hp.Size = Vector2.new(2, h * frac)
                        e.hp.Visible = true
                    else e.hp.Visible = false end
                    if C.esp_skel then
                        local bones = hum.RigType == Enum.HumanoidRigType.R15 and SKEL_R15 or SKEL_R6
                        local idx = 1
                        for _, pr in ipairs(bones) do
                            local a = char:FindFirstChild(pr[1])
                            local b = pr[2] and char:FindFirstChild(pr[2]) or nil
                            if a then
                                local p1 = CAM:WorldToViewportPoint(a.Position)
                                local p2 = b and CAM:WorldToViewportPoint(b.Position) or CAM:WorldToViewportPoint(a.Position + Vector3.new(0, -1, 0))
                                if p1.Z > 0 and p2.Z > 0 and e.skel[idx] then
                                    e.skel[idx].From = Vector2.new(p1.X, p1.Y)
                                    e.skel[idx].To = Vector2.new(p2.X, p2.Y)
                                    e.skel[idx].Color = C.esp_skel_col
                                    e.skel[idx].Visible = true
                                elseif e.skel[idx] then e.skel[idx].Visible = false end
                                idx = idx + 1
                            end
                        end
                        for i = idx, #e.skel do e.skel[i].Visible = false end
                    else
                        for _, l in ipairs(e.skel) do l.Visible = false end
                    end
                    esp_chams_apply(plr, char)
                else
                    e.box.Visible=false; e.name.Visible=false; e.dist.Visible=false; e.hp.Visible=false
                    for _, l in ipairs(e.skel) do l.Visible = false end
                    esp_chams_clear(plr)
                end
            end
        else
            if e then
                e.box.Visible=false; e.name.Visible=false; e.dist.Visible=false; e.hp.Visible=false
                for _, l in ipairs(e.skel) do l.Visible = false end
            end
            esp_chams_clear(plr)
        end
    end
end

    if rf and kb_backup.RF then pcall(function() rf.MeshId = kb_backup.RF.MeshId or ""; rf.Transparency = kb_backup.RF.Transparency or 0 end) end

local function headless_apply()

    local head = c:FindFirstChild("Head"); if not head then return end
    hl_backup = {trans = head.Transparency, mesh = head.MeshId, tex = head.TextureID}
    pcall(function()
        head.MeshId = "rbxassetid://6686307858"
        head.TextureID = "rbxassetid://6686307858"
        head.Transparency = 1
    end)
    for _, d in ipairs(head:GetDescendants()) do
        if d:IsA("BasePart") or d:IsA("Decal") then pcall(function() d.Transparency = 1 end) end
    end

        end)

-- fling
local function do_fling(tp)
    if not tp or not tp.Character then return end
    local hrp = my_hrp()
    local hum = LP.Character and LP.Character:FindFirstChildOfClass("Humanoid")
    if not hrp then return end
    local tc = tp.Character
    local thrp = tc:FindFirstChild("HumanoidRootPart") or tc:FindFirstChild("Head")
    local th = tc:FindFirstChildOfClass("Humanoid")
    if not thrp then return end
    getgenv().FLING_ACTIVE = (getgenv().FLING_ACTIVE or 0) + 1
    if hrp.Velocity.Magnitude < 50 then getgenv().OldPos = hrp.CFrame end
    if th and th.Sit then getgenv().FLING_ACTIVE = math.max(0, (getgenv().FLING_ACTIVE or 1) - 1); return end
    local old_fdh = WS.FallenPartsDestroyHeight
    pcall(function() WS.FallenPartsDestroyHeight = 0/0 end)
    local se = hum and hum:GetStateEnabled(Enum.HumanoidStateType.Seated)
    if hum then hum:SetStateEnabled(Enum.HumanoidStateType.Seated, false) end
    local tm = tick(); local ang = 0
    repeat
        if hrp and th then
            ang = ang + 100
            hrp.CFrame = CFrame.new(thrp.Position) * CFrame.new(0, 1.5, 0) * CFrame.Angles(math.rad(ang), 0, 0)
            hrp.Velocity = Vector3.new(9e7, 9e7*10, 9e7)
            hrp.RotVelocity = Vector3.new(9e8, 9e8, 9e8)
            task.wait()
        end
    until tm + 2 < tick() or not C.fling_targets
    if hum and se ~= nil then hum:SetStateEnabled(Enum.HumanoidStateType.Seated, se) end
    if getgenv().OldPos and hrp then
        hrp.CFrame = getgenv().OldPos * CFrame.new(0, 0.5, 0)
        if hum then hum:ChangeState("GettingUp") end
        pcall(function() WS.FallenPartsDestroyHeight = old_fdh end)
    end
    getgenv().FLING_ACTIVE = math.max(0, (getgenv().FLING_ACTIVE or 1) - 1)
end

local fling_thread = nil
local function start_fling_target()
    if fling_thread then return end
    fling_thread = task.spawn(function()
        while C.fling_targets do
            if TGT.player and TGT.player.Character and my_hrp() then do_fling(TGT.player) else task.wait(0.3) end
            task.wait()
        end
        fling_thread = nil
    end)
end

-- kill aura
local aura_thread = nil
local function get_knife()
    local c = LP.Character
    if c then local k = c:FindFirstChild("Knife"); if k then return k, true end end
    local bp = LP:FindFirstChildOfClass("Backpack")
    if bp then local k = bp:FindFirstChild("Knife"); if k then return k, false end end
end
local function knife_stab(k)
    local ev = k:FindFirstChild("Events"); local s = ev and ev:FindFirstChild("KnifeStabbed")
    if s then pcall(function() s:FireServer() end) end
end
local function knife_touch(k, part)
    local ev = k:FindFirstChild("Events"); local t = ev and ev:FindFirstChild("HandleTouched")
    if t then pcall(function() t:FireServer(part) end) end
end
local function aura_loop()
    while C.aura or C.kill_all do
        if my_role() == "Murderer" then
            local knife, eq = get_knife()
            if knife and not eq then
                local hum = LP.Character and LP.Character:FindFirstChildOfClass("Humanoid")
                if hum then pcall(function() hum:EquipTool(knife) end) end
            elseif knife then
                local myhrp = my_hrp()
                local victims = {}
                for _, plr in ipairs(Players:GetPlayers()) do
                    if plr == LP then continue end
                    local c = plr.Character
                    local hu = c and c:FindFirstChildOfClass("Humanoid")
                    local p = c and (c:FindFirstChild("HumanoidRootPart") or c:FindFirstChild("Head"))
                    if hu and hu.Health > 0 and p then
                        if C.kill_all then victims[#victims+1] = p
                        elseif C.aura and myhrp and (p.Position - myhrp.Position).Magnitude <= C.aura_dist then victims[#victims+1] = p end
                    end
                end
                if #victims > 0 then
                    knife_stab(knife)
                    for _, v in ipairs(victims) do knife_touch(knife, v) end
                end
            end
        end
        task.wait()
    end
    aura_thread = nil
end
local function start_aura()
    if not aura_thread then aura_thread = task.spawn(aura_loop) end
end

    M:slider_(atm_r, "fog end", 1000, 0, 5000, 0, function(v) C.fog_end = v; if C.fog then Lighting.FogEnd = v end end, "")
    M:toggle_(atm_r, "skybox", false, function(v) C.skybox_on = v; if v then apply_skybox(C.skybox) else clear_skybox() end end)
    M:dropdown_(atm_r, "preset", {"Jungle", "Blossom", "Red night"}, "Jungle", function(v) C.skybox = v; if C.skybox_on then apply_skybox(v) end end)

    -- ========================================
    -- MISC TAB
    -- ========================================
    local ms_t = M:tab({name = "misc", icon = TAB_ICONS.misc})
    local cross_l = M:section(ms_t, {name = "crosshair", side = "left"})

    M:color_(cross_l, "color", C.cross_col, function(c) C.cross_col = c end)

    local info_r = M:section(ms_t, {name = "info", side = "right"})
    M:label_(info_r, "SHITARO v6 standalone.\nAim: silent hook + prediction + backtrack.\nTracer: bullet trail.\nSelf chams: 5 modes.\nMenu: Insert or floating S on mobile.\nUnload: getgenv().SHITARO_UNLOAD()")

    notify("SHITARO v6 loaded. Insert to toggle.")
else
    warn("[SHITARO] menu failed:", M.err)
end

-- ============================================================
-- UNLOAD
-- ============================================================
getgenv().SHITARO_UNLOAD = function()
    C.aim=false; C.auto=false; C.force=false
    C.aura=false; C.kill_all=false; C.kill_targets=false; C.fling_targets=false
    C.fly=false; C.noclip=false; C.bhop=false; C.inf_jump=false
    C.fake=false; C.vs_on=false; C.esp=false; C.esp_chams=false
    C.self_chams=false; C.self_esp=false; C.tracer=false; C.cross=false
    C.spec=false; C.headsit=false; C.bang=false; C.looptp=false
    C.farm=false
    pcall(stop_fly); pcall(stop_fake); pcall(stop_bang)
    pcall(self_chams_restore); pcall(korblox_restore); pcall(headless_restore); pcall(clear_skybox)
    if se_hl then se_hl:Destroy() end
    for _, h in pairs(ESP_CHAMS) do pcall(function() h:Destroy() end) end
    table.clear(ESP_CHAMS)
    for plr, e in pairs(esp_cache) do
        for _, d in pairs(e) do
            if typeof(d) == "userdata" then pcall(function() d:Remove() end) end
        end
        for _, l in ipairs(e.skel or {}) do pcall(function() l:Remove() end) end
    end
    table.clear(esp_cache)
    for _, l in ipairs(cross_lines) do pcall(function() l:Remove() end) end
    if fov_circle then pcall(function() fov_circle:Remove() end) end
    if M.frame then pcall(function() M.frame:Destroy() end) end
    notify("SHITARO unloaded.")
end
