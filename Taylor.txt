-- ============================================================
-- attach gui
-- ============================================================
local function attach_gui(gui)
    if not gui then return nil end
    local candidates = {}
    if type(gethui) == "function" then
        local ok, h = pcall(gethui)
        if ok and h then candidates[#candidates+1] = h end
    end
    candidates[#candidates+1] = CoreGuiSvc
    local pg = LP:FindFirstChildOfClass("PlayerGui")
    if pg then candidates[#candidates+1] = pg end
    for i = 1, #candidates do
        local ok = pcall(function() gui.Parent = candidates[i] end)
        if ok and gui.Parent then return candidates[i] end
    end
end

function notify(text, dur)
    dur = dur or 3
    local gui = Instance.new("ScreenGui")
    gui.Name = "\0"
    gui.ResetOnSpawn = false
    gui.IgnoreGuiInset = true
    gui.DisplayOrder = 1000
    if not attach_gui(gui) then return end

    local f = Instance.new("Frame")
    f.BackgroundColor3 = Color3.fromRGB(22, 22, 26)
    f.BorderSizePixel = 0
    f.Size = UDim2.fromOffset(280, 50)
    f.Position = UDim2.new(1, -300, 0, 60)
    f.Parent = gui
    local uc = Instance.new("UICorner"); uc.CornerRadius = UDim.new(0, 8); uc.Parent = f
    local st = Instance.new("UIStroke"); st.Color = Color3.fromRGB(0, 170, 255); st.Thickness = 1.5; st.Parent = f

    local t = Instance.new("TextLabel")
    t.BackgroundTransparency = 1
    t.Size = UDim2.new(1, -20, 1, 0)
    t.Position = UDim2.fromOffset(10, 0)
    t.Font = Enum.Font.Gotham
    t.Text = text
    t.TextColor3 = Color3.fromRGB(230, 230, 240)
    t.TextSize = 13
    t.TextXAlignment = Enum.TextXAlignment.Left
    t.TextWrapped = true
    t.Parent = f

    pcall(function()
        TweenService:Create(f, TweenInfo.new(0.2), {Position = UDim2.new(1, -300, 0, 70)}):Play()
    end)
    task.delay(dur, function()
        pcall(function()
            TweenService:Create(f, TweenInfo.new(0.3), {BackgroundTransparency = 1}):Play()
            TweenService:Create(t, TweenInfo.new(0.3), {TextTransparency = 1}):Play()
        end)
        task.wait(0.4)
        pcall(function() gui:Destroy() end)
    end)
end

local UI = {}
UI.__index = UI

function UI.new(title)
    local self = setmetatable({}, UI)
    self.title = title or "SHITARO"
    self.enabled = true
    self.key = Enum.KeyCode.RightShift
    self.tabs = {}
    self.tab = nil
    self.mobile = UserInputService.TouchEnabled and true or false
    if self.mobile then
        self.size = Vector2.new(340, 480); self.kw = 58; self.kh = 32
    else
        self.size = Vector2.new(560, 420); self.kw = 40; self.kh = 20
    end
    local vp = CAM and CAM.ViewportSize or Vector2.new(800, 600)
    if self.mobile then
        self.pos = Vector2.new(math.max(10, (vp.X - self.size.X) / 2), 40)
    else
        self.pos = Vector2.new(60, 60)
    end
    self.ok, self.err = pcall(function() self:_build() end)
    if not self.ok then warn("[SHITARO] UI:", self.err) end
    return self
end

function UI:toggle(state)
    if not self.win then return end
    self.enabled = state and true or not self.enabled
    self.win.Visible = self.enabled
end

function UI:addTab(name)
    if not self.ok then return nil end
    local tab = {name=name, btn=nil, page=nil}
    local h = self.mobile and 44 or 28
    local b = Instance.new("TextButton")
    b.BackgroundColor3 = Color3.fromRGB(24, 24, 28)
    b.BorderSizePixel = 0
    b.Size = UDim2.new(1, 0, 0, h)
    b.Font = Enum.Font.GothamBold
    b.Text = name
    b.TextColor3 = Color3.fromRGB(200, 200, 210)
    b.TextSize = self.mobile and 15 or 13
    b.Parent = self.ts
    local uc = Instance.new("UICorner"); uc.CornerRadius = UDim.new(0, 6); uc.Parent = b
    tab.btn = b
    self.tabs[#self.tabs+1] = tab
    tab.page = self:_page(tab)
    tab.page.Visible = false
    b.MouseButton1Click:Connect(function() self:selectTab(tab) end)
    if #self.tabs == 1 then self:selectTab(tab) end
    return tab
end

function UI:selectTab(tab)
    if not tab or not tab.page then return end
    self.tab = tab
    for _, t in ipairs(self.tabs) do
        if t == tab then
            t.btn.BackgroundColor3 = Color3.fromRGB(0, 170, 255)
            t.btn.TextColor3 = Color3.fromRGB(255, 255, 255)
            t.page.Visible = true
        else
            t.btn.BackgroundColor3 = Color3.fromRGB(24, 24, 28)
            t.btn.TextColor3 = Color3.fromRGB(160, 160, 170)
            t.page.Visible = false
        end
    end
end

function UI:slider_(tab, sec, name, def, mn, mx, rnd, cb, suf)
    if not self.ok or not tab or not tab.page then return end
    local c = sec or tab.page.Root
    local rh = self.mobile and 68 or 42
    local r = Instance.new("Frame"); r.BackgroundTransparency = 1; r.Size = UDim2.new(1, 0, 0, rh); r.Parent = c
    local l = Instance.new("TextLabel")
    l.BackgroundTransparency = 1
    l.Size = UDim2.new(1, -70, 0, 20)
    l.Font = Enum.Font.Gotham
    l.Text = name
    l.TextColor3 = Color3.fromRGB(220, 220, 230)
    l.TextSize = self.mobile and 15 or 13
    l.TextXAlignment = Enum.TextXAlignment.Left
    l.Parent = r
    local v = Instance.new("TextLabel")
    v.BackgroundTransparency = 1
    v.Size = UDim2.fromOffset(70, 20)
    v.Position = UDim2.new(1, -70, 0, 0)
    v.Font = Enum.Font.GothamBold
    v.Text = tostring(def) .. (suf or "")
    v.TextColor3 = Color3.fromRGB(0, 170, 255)
    v.TextSize = self.mobile and 14 or 12
    v.TextXAlignment = Enum.TextXAlignment.Right
    v.Parent = r
    local by = self.mobile and 34 or 26
    local bh = self.mobile and 14 or 6
    local bar = Instance.new("Frame")
    bar.BackgroundColor3 = Color3.fromRGB(40, 40, 48)
    bar.BorderSizePixel = 0
    bar.Size = UDim2.new(1, 0, 0, bh)
    bar.Position = UDim2.new(0, 0, 0, by)
    bar.Parent = r
    local buc = Instance.new("UICorner"); buc.CornerRadius = UDim.new(1, 0); buc.Parent = bar
    local fill = Instance.new("Frame")
    fill.BackgroundColor3 = Color3.fromRGB(0, 170, 255)
    fill.BorderSizePixel = 0
    fill.Size = UDim2.fromScale((def - mn) / (mx - mn), 1)
    fill.Parent = bar
    local fuc = Instance.new("UICorner"); fuc.CornerRadius = UDim.new(1, 0); fuc.Parent = fill
    local drag = false
    local function setFromX(x)
        local a = bar.AbsolutePosition.X
        local s = bar.AbsoluteSize.X
        if s <= 0 then return end
        local p = math.clamp((x - a) / s, 0, 1)
        local raw = mn + (mx - mn) * p
        if rnd then
            local mult = 10 ^ rnd
            raw = math.floor(raw * mult + 0.5) / mult
        else
            raw = math.floor(raw + 0.5)
        end
        fill.Size = UDim2.fromScale((raw - mn) / (mx - mn), 1)
        v.Text = tostring(raw) .. (suf or "")
        if cb then pcall(cb, raw) end
    end
    local hit = Instance.new("TextButton")
    hit.BackgroundTransparency = 1
    hit.Text = ""
    hit.Size = UDim2.new(1, 0, 0, bh + (self.mobile and 24 or 8))
    hit.Position = UDim2.new(0, 0, 0, by - (self.mobile and 12 or 4))
    hit.Parent = r
    hit.InputBegan:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
            drag = true; setFromX(i.Position.X)
        end
    end)
    UserInputService.InputChanged:Connect(function(i)
        if drag and (i.UserInputType == Enum.UserInputType.MouseMovement or i.UserInputType == Enum.UserInputType.Touch) then
            setFromX(i.Position.X)
        end
    end)
    UserInputService.InputEnded:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then drag = false end
    end)
end

local BT = {hist={}, keep=2.0}
local function bt_track()
    local now = os.clock()
    local co = now - BT.keep
    for _, plr in ipairs(Players:GetPlayers()) do
        if plr == LP then continue end
        local char = plr.Character
        local hrp = char and char:FindFirstChild("HumanoidRootPart")
        local hum = char and char:FindFirstChildOfClass("Humanoid")
        if hrp and hum and hum.Health > 0 then
            local h = BT.hist[plr.Name]
            if not h then h = {}; BT.hist[plr.Name] = h end
            h[#h+1] = {t=now, pos=hrp.Position}
            local i = 1
            while i <= #h and h[i].t < co do i = i + 1 end
            if i > 1 then
                for j = 1, #h - (i - 1) do h[j] = h[j + i - 1] end
                for j = #h - (i - 1) + 1, #h do h[j] = nil end
            end
        end
    end
    for n in pairs(BT.hist) do
        if not Players:FindFirstChild(n) then BT.hist[n] = nil end
    end
end

local function bt_get(name, ms)
    local h = BT.hist[name]
    if not h or #h == 0 then return nil end
    local target = os.clock() - (ms / 1000)
    local best, bd = h[1], math.huge
    for i = 1, #h do
        local d = math.abs(h[i].t - target)
        if d < bd then best, bd = h[i], d end
    end
    return best and best.pos
end

local TGT = {player=nil, char=nil, part=nil, hum=nil}

local function holds(c, n) return c and c:FindFirstChild(n) ~= nil end

local function refresh_target()
    local found = nil
    if C.priority == "murderer" then
        local m = get_round()
        local d = m and m.PlayerData
        if type(d) == "table" then
            for name, info in pairs(d) do
                if type(info) == "table" and info.Role == "Murderer" and not info.Dead then
                    found = Players:FindFirstChild(name)
                    if found then break end
                end
            end
        end
        if not found then
            for _, plr in ipairs(Players:GetPlayers()) do
                if plr ~= LP and holds(plr.Character, "Knife") then found = plr; break end
            end
        end
    end
    if not found then
        local myhrp = LP.Character and LP.Character:FindFirstChild("HumanoidRootPart")
        if myhrp then
            local best, bd = nil, math.huge
            for _, plr in ipairs(Players:GetPlayers()) do
                if plr ~= LP and plr.Character then
                    local h = plr.Character:FindFirstChild("HumanoidRootPart")
                    local hu = plr.Character:FindFirstChildOfClass("Humanoid")
                    if h and hu and hu.Health > 0 then
                        local d = (h.Position - myhrp.Position).Magnitude
                        if d < bd and d <= C.range then best, bd = plr, d end
                    end
                end
            end
            found = best
        end
    end

    if found ~= TGT.player then
        TGT.player = found; TGT.char = nil; TGT.part = nil; TGT.hum = nil
    end
    if not found then return end
    local char = found.Character
    if char ~= TGT.char then TGT.char = char; TGT.part = nil; TGT.hum = nil end
    if not char then return end
    if not TGT.part or not TGT.part.Parent then
        TGT.part = char:FindFirstChild("HumanoidRootPart") or char:FindFirstChild("UpperTorso") or char:FindFirstChild("Torso")
    end
    if not TGT.hum or not TGT.hum.Parent then
        TGT.hum = char:FindFirstChildOfClass("Humanoid")
    end
end

local function tgt_alive()
    if not TGT.part or not TGT.part.Parent then return false end
    if not TGT.hum or not TGT.hum.Parent then return false end
    return TGT.hum.Health > 0
end

-- ============================================================
-- motion
-- ============================================================
local SN = {n=48, i=0, cnt=0, t={}, p={}, last=nil, lt=0, vel=Vector3.zero}
for i = 1, SN.n do SN.t[i] = 0; SN.p[i] = Vector3.zero end

local function sn_push(now, pos)
    SN.i = SN.i % SN.n + 1
    SN.t[SN.i] = now; SN.p[SN.i] = pos
    if SN.cnt < SN.n then SN.cnt = SN.cnt + 1 end
end

local function sn_get(k)
    local idx = (SN.i - k - 1) % SN.n + 1
    return SN.t[idx], SN.p[idx]
end

local function fit_vel()
    if SN.cnt < 3 then return nil end
    local newest = sn_get(0)
    local used, sd = 0, 0
    for k = 0, SN.cnt - 1 do
        local t = sn_get(k)
        if newest - t > 0.4 then break end
        used = used + 1; sd = sd + (t - newest)
    end
    if used < 3 then return nil end
    local md = sd / used
    local num, den = Vector3.zero, 0
    for k = 0, used - 1 do
        local t, p = sn_get(k)
        local d = (t - newest) - md
        num = num + p * d; den = den + d * d
    end
    if den < 1e-8 then return nil end
    return num / den
end

local function recent_vel()
    if SN.cnt < 2 then return nil end
    local newest, head = sn_get(0)
    local best = nil
    for k = 1, SN.cnt - 1 do
        local t, p = sn_get(k)
        local dt = newest - t
        if dt > 0.35 then break end
        best = (head - p) / dt
    end
    return best
end

local function track_motion(now)
    if not TGT.part or not TGT.part.Parent then SN.cnt=0; SN.i=0; SN.last=nil; return end
    local pos = TGT.part.Position
    if not SN.last then SN.last = pos; SN.lt = now; sn_push(now, pos); return end
    if now - SN.lt < 0.005 then return end
    SN.lt = now; SN.last = pos
    sn_push(now, pos)
    local fit = fit_vel()
    local fast = recent_vel()
    if fit and fast then
        local sf = Vector3.new(fit.X, 0, fit.Z).Magnitude
        local ss = Vector3.new(fast.X, 0, fast.Z).Magnitude
        if sf > 200 or sf ~= sf then fit = nil end
        if ss > 200 or ss ~= ss then fast = nil end
    end
    if fit and fast then
        SN.vel = Vector3.new(fast.X*0.7 + fit.X*0.3, fast.Y, fast.Z*0.7 + fit.Z*0.3)
    elseif fast then SN.vel = fast
    elseif fit then SN.vel = fit end
    local h = Vector3.new(SN.vel.X, 0, SN.vel.Z)
    if h.Magnitude > 120 then SN.vel = Vector3.new(SN.vel.X*0.4, SN.vel.Y, SN.vel.Z*0.4) end
end

local function clear_motion()
    SN.cnt=0; SN.i=0; SN.last=nil; SN.lt=0; SN.vel=Vector3.zero
end

local HIT = {"HumanoidRootPart","UpperTorso","Torso","LowerTorso","Head",
    "RightUpperArm","LeftUpperArm","RightUpperLeg","LeftUpperLeg"}

local function aim_point()
    if not TGT.char or not TGT.char.Parent then return nil end
    if not TGT.hum or TGT.hum.Health <= 0 then return nil end
    if C.backtrack then
        local b = bt_get(TGT.player and TGT.player.Name or "", C.bt_win)
        if b then return b end
    end
    local part = TGT.part
    if not part then
        for _, n in ipairs(HIT) do local p = TGT.char:FindFirstChild(n); if p then part = p; break end end
    end
    if not part then return nil end
    local base = part.Position
    if not C.predict then return base end
    if not PING.seen then return base end
    local t = math.clamp(PING.rtt, 0, 0.5)
    if t <= 0 then return base end
    local v = SN.vel
    local x = base.X + v.X * t
    local z = base.Z + v.Z * t
    local y = base.Y
    if math.abs(v.Y) > 1 then
        y = base.Y + v.Y * t - 0.5 * workspace.Gravity * t * t
    end
    return Vector3.new(x, y, z)
end

local force_att = nil
local force_orig = nil
local force_stamp = 0

local function restore_force()
    if not force_att then return end
    local a = force_att
    local o = force_orig
    force_att = nil; force_orig = nil
    if o then pcall(function() if a.Parent then a.CFrame = o end end) end
end

local function push_origin(cf)
    local c = LP.Character
    local hrp = c and c:FindFirstChild("HumanoidRootPart")
    if not hrp then return false end
    local att = hrp:FindFirstChild("GunRaycastAttachment")
    if not att then return false end
    if force_att and force_att ~= att then restore_force() end
    if not force_att then
        local ok, saved = pcall(function() return att.CFrame end)
        if not ok or typeof(saved) ~= "CFrame" then return false end
        force_att = att; force_orig = saved
    end
    force_stamp = os.clock()
    local ok = pcall(function() att.WorldCFrame = cf end)
    if not ok then restore_force(); return false end
    task.defer(restore_force)
    return true
end

local RAY = RaycastParams.new()
RAY.FilterType = Enum.RaycastFilterType.Exclude
RAY.IgnoreWater = false

local function origin_cf()
    local c = LP.Character
    local hrp = c and c:FindFirstChild("HumanoidRootPart")
    if not hrp then return nil end
    local att = hrp:FindFirstChild("GunRaycastAttachment")
    return att and att.WorldCFrame or hrp.CFrame
end

local function resolve_force()
    if not TGT.part or not TGT.part.Parent then return nil, nil end
    local live = TGT.part.Position
    local v = SN.vel
    local t = math.clamp(PING.rtt, 0, 0.5)
    local aim = C.predict and Vector3.new(
        live.X + v.X*t, live.Y + (math.abs(v.Y) > 1 and (v.Y*t - 0.5*workspace.Gravity*t*t) or 0), live.Z + v.Z*t
    ) or live
    local back = live - (aim - live).Unit * C.stand_off
    return CFrame.new(back, aim), CFrame.new(aim)
end

local function get_gun()
    local c = LP.Character
    if c then local g = c:FindFirstChild("Gun"); if g then return g, true end end
    local bp = LP:FindFirstChildOfClass("Backpack")
    if bp then local g = bp:FindFirstChild("Gun"); if g then return g, false end end

local last_shot = 0
local function auto_fire(now)
    if not C.auto or not C.aim or not tgt_alive() then return end
    if not get_gun() then return end
    local role = my_role()
    if role ~= "Sheriff" and role ~= "Hero" then
        -- fallback: allow if we hold a gun, game may lag behind role sync
        local gun = get_gun()
        if not gun then return end
    end
    local gun, eq = get_gun()
    if not gun then return end
    if not eq then
        local hum = LP.Character and LP.Character:FindFirstChildOfClass("Humanoid")
        if hum then pcall(function() hum:EquipTool(gun) end) end
        return
    end
    local delay = math.max(C.auto_delay, 0.05)
    if now - last_shot < delay then return end
    local p = aim_point()
    if not p then return end
    local remote = gun:FindFirstChild("Shoot")
    if not remote then
        for _, d in ipairs(gun:GetDescendants()) do
            if d:IsA("RemoteEvent") then remote = d; break end
        end
    end
    if not remote or not remote:IsA("RemoteEvent") then return end
    if C.force then
        local oc, ac = resolve_force()
        if oc and ac then
            if push_origin(oc) then
                pcall(function() remote:FireServer(oc, ac) end)
                last_shot = now
                return
            end
        end
    end
    local oc = origin_cf()
    if oc then
        pcall(function() remote:FireServer(oc, CFrame.new(p)) end)
        last_shot = now
    end
end

local function make_tracer(a, b)
    local dur = C.tracer_dur
    local pa = Instance.new("Part")
    pa.Transparency = 1; pa.Anchored = true; pa.CanCollide = false; pa.CanQuery = false
    pa.Size = Vector3.new(1,1,1); pa.CFrame = CFrame.new(a)
    Instance.new("Attachment", pa); pa.Parent = WS
    local pb = Instance.new("Part")
    pb.Transparency = 1; pb.Anchored = true; pb.CanCollide = false; pb.CanQuery = false
    pb.Size = Vector3.new(1,1,1); pb.CFrame = CFrame.new(b)
    Instance.new("Attachment", pb); pb.Parent = WS
    local beam = Instance.new("Beam")
    beam.FaceCamera = true
    beam.TextureSpeed = 1.5; beam.TextureLength = 2
    beam.Width0 = 0.25; beam.Width1 = 0.25
    beam.LightEmission = 3; beam.LightInfluence = 0; beam.Brightness = 2.5
    beam.Texture = "rbxassetid://12781800668"
    beam.Color = ColorSequence.new(C.tracer_col)
    beam.Transparency = NumberSequence.new({
        NumberSequenceKeypoint.new(0, 0.1), NumberSequenceKeypoint.new(1, 0.1)
    })
    beam.Attachment0 = pa.Attachment; beam.Attachment1 = pb.Attachment
    beam.Parent = pa
    task.delay(dur, function()
        pcall(function()
            if beam.Parent then
                TweenService:Create(beam, TweenInfo.new(0.2, Enum.EasingStyle.Linear), {Width0=0, Width1=0}):Play()
            end
        end)
    end)
    Debris:AddItem(pa, dur + 0.5); Debris:AddItem(pb, dur + 0.5)

local function connect_tracer()
    if tracer_remote and tracer_conn and tracer_conn.Connected then return end
    local ok, remote = pcall(function()
        return ReplicatedStorage:WaitForChild("ClientServices", 3):WaitForChild("WeaponService", 3):WaitForChild("GunFired", 3)
    end)
    if not ok or not remote then return end
    if tracer_conn then pcall(function() tracer_conn:Disconnect() end) end
    tracer_remote = remote
    tracer_conn = remote.OnClientEvent:Connect(function(gun, a, b)
        if not C.tracer then return end
        local c = LP.Character
        if typeof(gun) == "Instance" and c and gun:IsDescendantOf(c) then
            local pa, pb = to_pos(a), to_pos(b)
            if pa and pb then task.spawn(make_tracer, pa, pb) end
        end
    end)
end

local sc_backup = {}
local sc_hl = nil
local function self_chams_apply()
    local c = LP.Character
    if not c then return end
    if C.self_chams_type == "Highlight" then
        if not sc_hl or sc_hl.Parent ~= c then
            if sc_hl then sc_hl:Destroy() end
            sc_hl = Instance.new("Highlight")
            sc_hl.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
            sc_hl.FillTransparency = 0.5
            sc_hl.OutlineTransparency = 0
            sc_hl.Adornee = c
            sc_hl.Parent = c
        end
        sc_hl.FillColor = C.self_col
        sc_hl.OutlineColor = C.self_col
        sc_hl.Enabled = true
    else
        if sc_hl then sc_hl:Destroy() sc_hl = nil end
        for _, d in ipairs(c:GetDescendants()) do
            if d:IsA("BasePart") and d.Name ~= "HumanoidRootPart" then
                if sc_backup[d] == nil then
                    sc_backup[d] = {d.Material, d.Color, d.Transparency}
                end
                if C.self_chams_type == "ForceField" then
                    d.Material = Enum.Material.ForceField; d.Color = C.self_col
                elseif C.self_chams_type == "Neon" then
                    d.Material = Enum.Material.Neon; d.Color = C.self_col
                elseif C.self_chams_type == "Flat" then
                    d.Material = Enum.Material.SmoothPlastic; d.Color = C.self_col
                elseif C.self_chams_type == "Glass" then
                    d.Material = Enum.Material.Glass; d.Transparency = 0.5; d.Color = C.self_col
                end
            end
        end
    end
end

local function self_chams_restore()
    for p, d in pairs(sc_backup) do
        if p and p.Parent then
            pcall(function() p.Material = d[1]; p.Color = d[2]; p.Transparency = d[3] end)
        end
    end
    sc_backup = {}
    if sc_hl then sc_hl:Destroy() sc_hl = nil end
end

local esp_cache = {}

local function new_esp(plr)
    if esp_cache[plr] then return esp_cache[plr] end
    if not HAS_DR then return nil end
    local e = {
        box = Drawing.new("Square"),
        name = Drawing.new("Text"),
        dist = Drawing.new("Text"),
        hp = Drawing.new("Square"),
        skel = {},
    }
    e.box.Thickness = 1; e.box.Filled = false; e.box.Transparency = 1; e.box.Visible = false
    e.name.Center = true; e.name.Outline = true; e.name.Size = 13; e.name.Font = Drawing.Fonts.UI; e.name.Visible = false
    e.dist.Center = true; e.dist.Outline = true; e.dist.Size = 12; e.dist.Font = Drawing.Fonts.UI; e.dist.Visible = false
    e.hp.Filled = true; e.hp.Thickness = 0; e.hp.Transparency = 1; e.hp.Visible = false
    for i = 1, 20 do
        local l = Drawing.new("Line")
        l.Thickness = 1; l.Transparency = 1; l.Visible = false
        e.skel[i] = l
    end
    esp_cache[plr] = e
    return e
end

local function role_color(plr)
    local r = target_role(plr)
    if r == "Murderer" then return C.esp_col_mur end
    if r == "Sheriff" then return C.esp_col_shf end
    return C.esp_col_inn
end

local function esp_update()
    if not HAS_DR then return end
    if not C.esp then
        for plr, e in pairs(esp_cache) do
            e.box.Visible = false; e.name.Visible = false; e.dist.Visible = false; e.hp.Visible = false
            for _, l in ipairs(e.skel) do l.Visible = false end
        end
        return
    end
    for _, plr in ipairs(Players:GetPlayers()) do
        if plr == LP then continue end
        local char = plr.Character
        local hrp = char and char:FindFirstChild("HumanoidRootPart")
        local hum = char and char:FindFirstChildOfClass("Humanoid")
        local e = esp_cache[plr]
        if char and hrp and hum and hum.Health > 0 then
            if not e then e = new_esp(plr) end
            if e then
                local top = CAM:WorldToViewportPoint(hrp.Position + Vector3.new(0, 3, 0))
                local bot = CAM:WorldToViewportPoint(hrp.Position - Vector3.new(0, 3, 0))
                if top.Z > 0 and bot.Z > 0 then
                    local h = math.abs(top.Y - bot.Y)
                    local w = h * 0.55
                    local bx = bot.X - w / 2
                    local by = top.Y
                    if C.esp_box then
                        e.box.Color = role_color(plr)
                        e.box.Position = Vector2.new(bx, by)
                        e.box.Size = Vector2.new(w, h)
                        e.box.Visible = true
                    else e.box.Visible = false end
                    if C.esp_name then
                        e.name.Color = C.esp_name_col
                        e.name.Text = plr.DisplayName
                        e.name.Position = Vector2.new(bot.X, by - 16)
                        e.name.Visible = true
                    else e.name.Visible = false end
                    if C.esp_dist then
                        local myhrp = LP.Character and LP.Character:FindFirstChild("HumanoidRootPart")
                        local d = myhrp and (hrp.Position - myhrp.Position).Magnitude or 0
                        e.dist.Color = C.esp_dist_col
                        e.dist.Text = string.format("[%d]", math.floor(d + 0.5))
                        e.dist.Position = Vector2.new(bot.X, bot.Y + 2)
                        e.dist.Visible = true
                    else e.dist.Visible = false end
                    if C.esp_hp then
                        local frac = math.clamp(hum.Health / math.max(hum.MaxHealth, 1), 0, 1)
                        e.hp.Color = C.esp_hp_col
                        e.hp.Position = Vector2.new(bx - 5, by + h * (1 - frac))
                        e.hp.Size = Vector2.new(2, h * frac)
                        e.hp.Visible = true
                    else e.hp.Visible = false end
                    if C.esp_skel then
                        local bones = hum.RigType == Enum.HumanoidRigType.R15 and SKEL_BONES or R6_SKEL
                        local idx = 1
                        for _, pair in ipairs(bones) do
                            local a = char:FindFirstChild(pair[1])
                            local b = pair[2] and char:FindFirstChild(pair[2]) or nil
                            if a and (b or not pair[2]) then
                                local p1 = CAM:WorldToViewportPoint(a.Position)
                                local p2 = b and CAM:WorldToViewportPoint(b.Position) or CAM:WorldToViewportPoint(a.Position + Vector3.new(0, -1, 0))
                                if p1.Z > 0 and p2.Z > 0 and e.skel[idx] then
                                    e.skel[idx].From = Vector2.new(p1.X, p1.Y)
                                    e.skel[idx].To = Vector2.new(p2.X, p2.Y)
                                    e.skel[idx].Color = C.esp_skel_col
                                    e.skel[idx].Visible = true
                                elseif e.skel[idx] then
                                    e.skel[idx].Visible = false
                                end
                                idx = idx + 1
                            end
                        end
                        for i = idx, #e.skel do e.skel[i].Visible = false end
                    else
                        for _, l in ipairs(e.skel) do l.Visible = false end
                    end
                    esp_chams_apply(plr, char)
                else
                    e.box.Visible = false; e.name.Visible = false; e.dist.Visible = false; e.hp.Visible = false
                    for _, l in ipairs(e.skel) do l.Visible = false end
                    esp_chams_clear(plr)
                end
            end
        else
            if e then
                e.box.Visible = false; e.name.Visible = false; e.dist.Visible = false; e.hp.Visible = false
                for _, l in ipairs(e.skel) do l.Visible = false end
            end
            esp_chams_clear(plr)
        end
    end
end

local aura_thread = nil
local function get_knife()
    local c = LP.Character
    if c then local k = c:FindFirstChild("Knife"); if k then return k, true end end
    local bp = LP:FindFirstChildOfClass("Backpack")
    if bp then local k = bp:FindFirstChild("Knife"); if k then return k, false end end
end

local function knife_stab(knife)
    local ev = knife:FindFirstChild("Events")
    local s = ev and ev:FindFirstChild("KnifeStabbed")
    if s then pcall(function() s:FireServer() end) end
end

local function knife_touch(knife, part)
    local ev = knife:FindFirstChild("Events")
    local t = ev and ev:FindFirstChild("HandleTouched")
    if t then pcall(function() t:FireServer(part) end) end
end

local function aura_loop()
    while C.aura or C.kill_all do
        local role = my_role()
        if role == "Murderer" then
            local knife, eq = get_knife()
            if knife and not eq then
                local hum = LP.Character and LP.Character:FindFirstChildOfClass("Humanoid")
                if hum then pcall(function() hum:EquipTool(knife) end) end
            elseif knife then
                local myhrp = LP.Character and LP.Character:FindFirstChild("HumanoidRootPart")
                local victims = {}
                for _, plr in ipairs(Players:GetPlayers()) do
                    if plr == LP then continue end
                    local c = plr.Character
                    local hu = c and c:FindFirstChildOfClass("Humanoid")
                    local p = c and (c:FindFirstChild("HumanoidRootPart") or c:FindFirstChild("Head"))
                    if hu and hu.Health > 0 and p then
                        if C.kill_all then
                            victims[#victims+1] = p
                        elseif C.aura and myhrp and (p.Position - myhrp.Position).Magnitude <= C.aura_dist then
                            victims[#victims+1] = p
                        end
                    end
                end
                if #victims > 0 then
                    knife_stab(knife)
                    for _, v in ipairs(victims) do knife_touch(knife, v) end
                end
            end
        end
        task.wait()
    end
    aura_thread = nil
end

local function start_aura()
    if not aura_thread then aura_thread = task.spawn(aura_loop) end
end

        for i = 1, #cross_lines do cross_lines[i].Visible = false end
        return
    end
    cross_build()
    local m = UserInputService:GetMouseLocation()
    local cx, cy = m.X, m.Y + 36
    for i = 1, 4 do
        local ang = (i - 1) * math.pi / 2
        local idx = (i - 1) * 2 + 1
        local out = idx + 1
        cross_lines[idx].Color = C.cross_col
        cross_lines[idx].Thickness = C.cross_th
        cross_lines[out].Color = Color3.new(0, 0, 0)
        cross_lines[out].Thickness = C.cross_th + 2
        local sx = cx + C.cross_gap * math.cos(ang)
        local sy = cy + C.cross_gap * math.sin(ang)
        local ex = cx + (C.cross_gap + C.cross_len) * math.cos(ang)
        local ey = cy + (C.cross_gap + C.cross_len) * math.sin(ang)
        cross_lines[idx].From = Vector2.new(sx, sy)
        cross_lines[idx].To = Vector2.new(ex, ey)
        cross_lines[idx].Visible = true
        cross_lines[out].From = Vector2.new(sx - 1, sy - 1)
        cross_lines[out].To = Vector2.new(ex + 1, ey + 1)
        cross_lines[out].Visible = true
    end
end

-- fling
local function do_fling(tp)
    if not tp or not tp.Character then return end
    local hrp = my_hrp()
    local hum = LP.Character and LP.Character:FindFirstChildOfClass("Humanoid")
    if not hrp then return end
    local tc = tp.Character
    local thrp = tc:FindFirstChild("HumanoidRootPart") or tc:FindFirstChild("Head")
    local th = tc:FindFirstChildOfClass("Humanoid")
    if not thrp then return end
    getgenv().FLING_ACTIVE = (getgenv().FLING_ACTIVE or 0) + 1
    if hrp.Velocity.Magnitude < 50 then getgenv().OldPos = hrp.CFrame end
    if th and th.Sit then
        getgenv().FLING_ACTIVE = math.max(0, (getgenv().FLING_ACTIVE or 1) - 1)
        return
    end
    local old_fdh = WS.FallenPartsDestroyHeight
    pcall(function() WS.FallenPartsDestroyHeight = 0/0 end)
    local se = hum and hum:GetStateEnabled(Enum.HumanoidStateType.Seated)
    if hum then hum:SetStateEnabled(Enum.HumanoidStateType.Seated, false) end
    local tm = tick()
    local ang = 0
    repeat
        if hrp and th then
            local tv = C.fling_vel and (th.MoveDirection * th.WalkSpeed) or thrp.Velocity
            ang = ang + 100
            hrp.CFrame = CFrame.new(thrp.Position) * CFrame.new(0, 1.5, 0) * CFrame.Angles(math.rad(ang), 0, 0)
            hrp.Velocity = Vector3.new(9e7, 9e7*10, 9e7)
            hrp.RotVelocity = Vector3.new(9e8, 9e8, 9e8)
            task.wait()
        end
    until tm + 2 < tick() or not C.fling_targets
    if hum and se ~= nil then hum:SetStateEnabled(Enum.HumanoidStateType.Seated, se) end
    if getgenv().OldPos and hrp then
        hrp.CFrame = getgenv().OldPos * CFrame.new(0, 0.5, 0)
        if hum then hum:ChangeState("GettingUp") end
        pcall(function() WS.FallenPartsDestroyHeight = old_fdh end)
    end
    getgenv().FLING_ACTIVE = math.max(0, (getgenv().FLING_ACTIVE or 1) - 1)
end

local fling_thread = nil
local function start_fling_target()
    if fling_thread then return end
    fling_thread = task.spawn(function()
        while C.fling_targets do
            if TGT.player and TGT.player.Character and my_hrp() then
                do_fling(TGT.player)
            else
                task.wait(0.3)
            end
            task.wait()
        end
        fling_thread = nil
    end)
end

    if kill_thread then return end

local farm_on = false
local farm_conn = nil
local farm_noclip_cache = {}
local function start_farm()
    if farm_conn then return end
    farm_conn = RunService.Stepped:Connect(function(_, dt)
        if not farm_on then
            for p, v in pairs(farm_noclip_cache) do
                if p and p.Parent then pcall(function() p.CanCollide = v end) end
            end
            table.clear(farm_noclip_cache)
            return
        end
        local c = LP.Character
        local hrp = c and c:FindFirstChild("HumanoidRootPart")
        local hu = c and c:FindFirstChildOfClass("Humanoid")
        if not hrp or not hu or hu.Health <= 0 then return end
        for _, p in ipairs(c:GetDescendants()) do
            if p:IsA("BasePart") and p.CanCollide then
                if farm_noclip_cache[p] == nil then farm_noclip_cache[p] = true end
                p.CanCollide = false
            end
        end
        local coins = CollectionService:GetTagged("CoinVisual")
        local best, bd = nil, math.huge
        for _, v in ipairs(coins) do
            if v and v.Parent and v:IsA("BasePart") then
                if not v:GetAttribute("Collected") and not v:GetAttribute("Delete") then
                    local d = (v.Position - hrp.Position).Magnitude
                    if d < bd then best, bd = v, d end
                end
            end
        end
        if best then
            local d = best.Position - hrp.Position
            local step = math.min(C.range, 23 * dt)
            local np = hrp.Position + (d.Magnitude > 0 and d.Unit * step or Vector3.zero)
            pcall(function()
                hrp.CFrame = CFrame.new(np)
                hrp.AssemblyLinearVelocity = Vector3.zero
            end)
            if (best.Position - hrp.Position).Magnitude <= 6 then
                if type(firetouchinterest) == "function" then
                    pcall(firetouchinterest, hrp, best, 0)
                    pcall(firetouchinterest, hrp, best, 1)
                end
            end
        end
    end)
end

    M:slider_(ms_t, ms_s, "gap", 4, 0, 20, 0, function(v) C.cross_gap = v end, "")
    M:slider_(ms_t, ms_s, "length", 8, 2, 30, 0, function(v) C.cross_len = v end, "")
    M:slider_(ms_t, ms_s, "thickness", 2, 1, 5, 0, function(v) C.cross_th = v end, "")
    M:color_(ms_t, ms_s, "color", C.cross_col, function(c) C.cross_col = c end)

    local tool_s = ms_t.page.NewSection("tools")
    M:button_(ms_t, tool_s, "tp tool", function() give_tp_tool(); notify("tp tool given") end)
    M:button_(ms_t, tool_s, "remove tp tool", function() remove_tp_tool() end)
    M:toggle_(ms_t, tool_s, "auto farm", false, function(v)
        farm_on = v
        if v then start_farm() end
    end)

    local alert_s = ms_t.page.NewSection("alerts")
    M:toggle_(ms_t, alert_s, "alerts", false, function(v) C.alert = v; if v then start_alerts() end end)
    M:toggle_(ms_t, alert_s, "miss", false, function(v) C.alert_miss = v end)
    M:toggle_(ms_t, alert_s, "kill murder", false, function(v) C.alert_kill = v end)

    local info_s = ms_t.page.NewSection("info")
    local info = Instance.new("TextLabel")
    info.BackgroundTransparency = 1
    info.Size = UDim2.new(1, 0, 0, 200)
    info.Font = Enum.Font.Gotham
    info.Text = "SHITARO v4\n\nDrawing: " .. tostring(HAS_DR) .. "\nMobile: " .. tostring(M.mobile) .. "\n\nPriority aim: murderer -> nearest\nForce shoot: corridor + attach override\nAuto: fires Shoot remote every delay\n\nUnload: getgenv().SHITARO_UNLOAD()"
    info.TextColor3 = Color3.fromRGB(180, 180, 190)
    info.TextSize = 12
    info.TextWrapped = true
    info.TextXAlignment = Enum.TextXAlignment.Left
    info.TextYAlignment = Enum.TextYAlignment.Top
    info.Parent = info_s

    notify("SHITARO v4 loaded. Menu open.")
else
    warn("[SHITARO] menu build failed:", M.err)
end

getgenv().SHITARO_UNLOAD = function()
    C.aim = false; C.auto = false; C.force = false
    C.aura = false; C.kill_all = false; C.kill_targets = false; C.fling_targets = false
    C.fly = false; C.noclip = false; C.bhop = false; C.inf_jump = false
    C.fake = false; C.vs_on = false
    C.esp = false; C.esp_chams = false
    C.self_chams = false; C.self_esp = false
    C.tracer = false
    C.cross = false
    C.spec = false; C.headsit = false; C.bang = false; C.looptp = false
    C.alert = false
    farm_on = false
    pcall(stop_fly)
    pcall(stop_fake)
    pcall(stop_bang)
    pcall(self_chams_restore)
    pcall(korblox_restore)
    pcall(headless_restore)
    pcall(clear_skybox)
    pcall(remove_tp_tool)
    if se_hl then se_hl:Destroy() end
    for _, h in pairs(ESP_CHAMS) do pcall(function() h:Destroy() end) end
    table.clear(ESP_CHAMS)
    for plr, e in pairs(esp_cache) do
        for _, d in pairs(e) do
            if typeof(d) == "userdata" then pcall(function() d:Remove() end) end
        end
        for _, l in ipairs(e.skel or {}) do pcall(function() l:Remove() end) end
    end
    table.clear(esp_cache)
    for _, l in ipairs(cross_lines) do pcall(function() l:Remove() end) end
    if fov_circle then pcall(function() fov_circle:Remove() end) end
    if M.frame then pcall(function() M.frame:Destroy() end) end
    notify("SHITARO unloaded.")
end
