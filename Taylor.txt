-- ============================================================
-- mobile ui kit (touch-friendly)
-- ============================================================
local UI = {}
UI.__index = UI

local function isTouch()
    return UserInputService.TouchEnabled
end

function UI.new(title)
    local self = setmetatable({}, UI)
    self.title = title or "SHITARO"
    self.enabled = true
    self.key = Enum.KeyCode.RightShift
    self.tabs = {}
    self.tab = nil
    self.scroll = 0
    self.mobile = isTouch()

    -- размеры под мобилу
    if self.mobile then
        self.size = Vector2.new(340, 480)
        self.tab_h = 60        -- высота таба (крупнее)
        self.row_h = 56        -- высота toggle-строки
        self.knob_w = 58       -- ширина switch
        self.knob_h = 32
    else
        self.size = Vector2.new(520, 400)
        self.tab_h = 30
        self.row_h = 28
        self.knob_w = 40
        self.knob_h = 20
    end

    -- стартовая позиция
    local vp = workspace.CurrentCamera and workspace.CurrentCamera.ViewportSize or Vector2.new(800, 600)
    if self.mobile then
        self.pos = Vector2.new(math.max(10, (vp.X - self.size.X) / 2), 40)
    else
        self.pos = Vector2.new(60, 60)
    end

    self.drag = nil
    self:build()
    return self
end

function UI:build()
    self.frame = Instance.new("ScreenGui")
    self.frame.Name = "\0"
    self.frame.ResetOnSpawn = false
    self.frame.IgnoreGuiInset = true
    self.frame.ZIndexBehavior = Enum.ZIndexBehavior.Sibling
    self.frame.DisplayOrder = 999
    pcall(function() self.frame.Parent = gethui and gethui() or CoreGui end)

    -- WINDOW
    self.win = Instance.new("Frame")
    self.win.Name = "\0"
    self.win.BackgroundColor3 = Color3.fromRGB(16, 16, 18)
    self.win.BorderSizePixel = 0
    self.win.Position = UDim2.fromOffset(self.pos.X, self.pos.Y)
    self.win.Size = UDim2.fromOffset(self.size.X, self.size.Y)
    self.win.Active = true
    self.win.Draggable = false
    self.win.Parent = self.frame

    local uc = Instance.new("UICorner")
    uc.CornerRadius = UDim.new(0, 10)
    uc.Parent = self.win

    local stroke = Instance.new("UIStroke")
    stroke.Color = Color3.fromRGB(70, 70, 80)
    stroke.Thickness = 1.5
    stroke.Parent = self.win

    -- TITLE BAR
    self.title_bar = Instance.new("Frame")
    self.title_bar.Name = "TitleBar"
    self.title_bar.BackgroundColor3 = Color3.fromRGB(22, 22, 26)
    self.title_bar.BorderSizePixel = 0
    self.title_bar.Size = UDim2.new(1, 0, 0, self.mobile and 48 or 32)
    self.title_bar.Active = true
    self.title_bar.Parent = self.win

    local tuc = Instance.new("UICorner")
    tuc.CornerRadius = UDim.new(0, 10)
    tuc.Parent = self.title_bar

    self.title_label = Instance.new("TextLabel")
    self.title_label.BackgroundTransparency = 1
    self.title_label.Size = UDim2.new(1, -70, 1, 0)
    self.title_label.Position = UDim2.fromOffset(14, 0)
    self.title_label.Font = Enum.Font.GothamBold
    self.title_label.Text = self.title
    self.title_label.TextColor3 = Color3.fromRGB(240, 240, 245)
    self.title_label.TextSize = self.mobile and 18 or 14
    self.title_label.TextXAlignment = Enum.TextXAlignment.Left
    self.title_label.Parent = self.title_bar

    -- close button (bigger on mobile)
    local closeSize = self.mobile and 44 or 28
    self.close_btn = Instance.new("TextButton")
    self.close_btn.BackgroundTransparency = 1
    self.close_btn.Size = UDim2.fromOffset(closeSize, closeSize)
    self.close_btn.Position = UDim2.new(1, -closeSize - 4, 0, (self.title_bar.AbsoluteSize.Y - closeSize) / 2)
    self.close_btn.Font = Enum.Font.GothamBold
    self.close_btn.Text = "×"
    self.close_btn.TextColor3 = Color3.fromRGB(220, 220, 220)
    self.close_btn.TextSize = self.mobile and 28 or 20
    self.close_btn.Parent = self.title_bar
    self.close_btn.MouseButton1Click:Connect(function()
        self:toggle(false)
    end)

    -- TAB STRIP
    self.tab_strip = Instance.new("Frame")
    self.tab_strip.BackgroundColor3 = Color3.fromRGB(20, 20, 24)
    self.tab_strip.BorderSizePixel = 0
    self.tab_strip.Size = UDim2.new(0, 110, 1, -(self.mobile and 48 or 32))
    self.tab_strip.Position = UDim2.fromOffset(0, self.mobile and 48 or 32)
    self.tab_strip.Parent = self.win

    local tsl = Instance.new("UIListLayout")
    tsl.Padding = UDim.new(0, 4)
    tsl.SortOrder = Enum.SortOrder.LayoutOrder
    tsl.Parent = self.tab_strip

    local tsp = Instance.new("UIPadding")
    tsp.PaddingTop = UDim.new(0, 6)
    tsp.PaddingLeft = UDim.new(0, 6)
    tsp.PaddingRight = UDim.new(0, 6)
    tsp.Parent = self.tab_strip

    -- BODY (scrolling frame, touch friendly)
    self.body = Instance.new("ScrollingFrame")
    self.body.BackgroundColor3 = Color3.fromRGB(16, 16, 18)
    self.body.BorderSizePixel = 0
    self.body.Size = UDim2.new(1, -110, 1, -(self.mobile and 48 or 32))
    self.body.Position = UDim2.fromOffset(110, self.mobile and 48 or 32)
    self.body.ScrollBarThickness = self.mobile and 8 or 4
    self.body.ScrollBarImageColor3 = Color3.fromRGB(90, 90, 110)
    self.body.CanvasSize = UDim2.new(0, 0, 0, 0)
    self.body.AutomaticCanvasSize = Enum.AutomaticSize.Y
    self.body.ScrollingDirection = Enum.ScrollingDirection.Y
    self.body.ElasticBehavior = Enum.ElasticBehavior.WhenScrollable
    self.body.Parent = self.win

    local bl = Instance.new("UIListLayout")
    bl.Padding = UDim.new(0, 10)
    bl.SortOrder = Enum.SortOrder.LayoutOrder
    bl.Parent = self.body

    local bp = Instance.new("UIPadding")
    bp.PaddingLeft = UDim.new(0, 12)
    bp.PaddingRight = UDim.new(0, 12)
    bp.PaddingTop = UDim.new(0, 12)
    bp.PaddingBottom = UDim.new(0, 20)
    bp.Parent = self.body

    -- DRAG (mouse + touch)
    local dragStart, startPos

    local function dragBegin(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1
            or input.UserInputType == Enum.UserInputType.Touch then
            dragStart = Vector2.new(input.Position.X, input.Position.Y)
            startPos = self.win.Position
            if self.mobile then
                self.win.Size = UDim2.fromOffset(self.size.X, self.size.Y)
            end
        end
    end

    local function dragMove(input)
        if not dragStart then return end
        if input.UserInputType ~= Enum.UserInputType.MouseMovement
            and input.UserInputType ~= Enum.UserInputType.Touch then return end
        local p = Vector2.new(input.Position.X, input.Position.Y)
        local delta = p - dragStart
        self.win.Position = UDim2.new(
            startPos.X.Scale, startPos.X.Offset + delta.X,
            startPos.Y.Scale, startPos.Y.Offset + delta.Y
        )
    end

    local function dragEnd(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1
            or input.UserInputType == Enum.UserInputType.Touch then
            dragStart = nil
        end
    end

    self.title_bar.InputBegan:Connect(dragBegin)
    self.title_bar.InputChanged:Connect(dragMove)
    self.title_bar.InputEnded:Connect(dragEnd)

    -- на мобиле — отдельная floating-кнопка открытия/закрытия
    if self.mobile then
        self.float_btn = Instance.new("TextButton")
        self.float_btn.BackgroundColor3 = Color3.fromRGB(0, 170, 255)
        self.float_btn.BorderSizePixel = 0
        self.float_btn.Size = UDim2.fromOffset(56, 56)
        self.float_btn.Position = UDim2.new(0, 20, 1, -90)
        self.float_btn.Font = Enum.Font.GothamBold
        self.float_btn.Text = "≡"
        self.float_btn.TextColor3 = Color3.fromRGB(255, 255, 255)
        self.float_btn.TextSize = 30
        self.float_btn.AutoButtonColor = true
        self.float_btn.Parent = self.frame
        local fuc = Instance.new("UICorner")
        fuc.CornerRadius = UDim.new(1, 0)
        fuc.Parent = self.float_btn

        self.float_btn.MouseButton1Click:Connect(function()
            self:toggle(not self.enabled)
        end)

        -- floating btn draggable
        local fbDragStart, fbStart
        self.float_btn.InputBegan:Connect(function(input)
            if input.UserInputType == Enum.UserInputType.Touch
                or input.UserInputType == Enum.UserInputType.MouseButton1 then
                fbDragStart = Vector2.new(input.Position.X, input.Position.Y)
                fbStart = self.float_btn.Position
            end
        end)
        self.float_btn.InputChanged:Connect(function(input)
            if not fbDragStart then return end
            if input.UserInputType == Enum.UserInputType.Touch
                or input.UserInputType == Enum.UserInputType.MouseMovement then
                local p = Vector2.new(input.Position.X, input.Position.Y)
                local d = p - fbDragStart
                self.float_btn.Position = UDim2.new(
                    fbStart.X.Scale, fbStart.X.Offset + d.X,
                    fbStart.Y.Scale, fbStart.Y.Offset + d.Y
                )
            end
        end)
        self.float_btn.InputEnded:Connect(function(input)
            if input.UserInputType == Enum.UserInputType.Touch
                or input.UserInputType == Enum.UserInputType.MouseButton1 then
                fbDragStart = nil
            end
        end)
    end

    -- keyboard shortcut (all platforms)
    UserInputService.InputBegan:Connect(function(input, gp)
        if gp then return end
        if input.KeyCode == self.key then
            self:toggle(not self.enabled)
        end
    end)

    -- default: open
    self.win.Visible = true
end

function UI:toggle(state)
    self.enabled = state and true or not self.enabled
    self.win.Visible = self.enabled
end

function UI:addTab(name)
    local tab = { name = name, items = {}, btn = nil, page = nil }
    local btnH = self.mobile and 44 or 28

    local btn = Instance.new("TextButton")
    btn.BackgroundColor3 = Color3.fromRGB(24, 24, 28)
    btn.BorderSizePixel = 0
    btn.Size = UDim2.new(1, 0, 0, btnH)
    btn.Font = Enum.Font.GothamBold
    btn.Text = name
    btn.TextColor3 = Color3.fromRGB(200, 200, 210)
    btn.TextSize = self.mobile and 15 or 13
    btn.TextXAlignment = Enum.TextXAlignment.Center
    btn.AutoButtonColor = true
    btn.Parent = self.tab_strip

    local buc = Instance.new("UICorner")
    buc.CornerRadius = UDim.new(0, 6)
    buc.Parent = btn

    tab.btn = btn
    self.tabs[#self.tabs + 1] = tab

    btn.MouseButton1Click:Connect(function()
        self:selectTab(tab)
    end)

    if #self.tabs == 1 then
        self:selectTab(tab)
    end
    return tab
end

function UI:selectTab(tab)
    self.tab = tab
    for _, t in ipairs(self.tabs) do
        if t == tab then
            t.btn.BackgroundColor3 = Color3.fromRGB(0, 170, 255)
            t.btn.TextColor3 = Color3.fromRGB(255, 255, 255)
        else
            t.btn.BackgroundColor3 = Color3.fromRGB(24, 24, 28)
            t.btn.TextColor3 = Color3.fromRGB(160, 160, 170)
        end
    end
    if not tab.page then
        tab.page = self:_buildPage(tab)
    end
    for _, t in ipairs(self.tabs) do
        if t.page then t.page.Visible = (t == tab) end
    end
end

function UI:_buildPage(tab)
    local page = Instance.new("Frame")
    page.BackgroundTransparency = 1
    page.Size = UDim2.fromScale(1, 1)
    page.Parent = self.body

    local ll = Instance.new("UIListLayout")
    ll.Padding = UDim.new(0, 12)
    ll.SortOrder = Enum.SortOrder.LayoutOrder
    ll.Parent = page

    local function newSection(name)
        local sec = Instance.new("Frame")
        sec.BackgroundColor3 = Color3.fromRGB(22, 22, 26)
        sec.BorderSizePixel = 0
        sec.Size = UDim2.new(1, 0, 0, 0)
        sec.AutomaticSize = Enum.AutomaticSize.Y
        sec.Parent = page
        local suc = Instance.new("UICorner")
        suc.CornerRadius = UDim.new(0, 8)
        suc.Parent = sec
        local pad = Instance.new("UIPadding")
        pad.PaddingLeft = UDim.new(0, 12)
        pad.PaddingRight = UDim.new(0, 12)
        pad.PaddingTop = UDim.new(0, 10)
        pad.PaddingBottom = UDim.new(0, 12)
        pad.Parent = sec

        local title = Instance.new("TextLabel")
        title.BackgroundTransparency = 1
        title.Size = UDim2.new(1, 0, 0, 18)
        title.Font = Enum.Font.GothamBold
        title.Text = string.upper(name)
        title.TextColor3 = Color3.fromRGB(140, 140, 150)
        title.TextSize = self.mobile and 12 or 11
        title.TextXAlignment = Enum.TextXAlignment.Left
        title.Parent = sec

        local ll2 = Instance.new("UIListLayout")
        ll2.Padding = UDim.new(0, 8)
        ll2.SortOrder = Enum.SortOrder.LayoutOrder
        ll2.Parent = sec
        local pad2 = Instance.new("UIPadding")
        pad2.PaddingTop = UDim.new(0, 22)
        pad2.Parent = sec
        return sec
    end

    return { Visible = false, Frame = page, NewSection = newSection }
end

function UI:addToggle(tab, section, name, default, cb, flag)
    local container = section or tab.page.Frame
    local rowH = self.mobile and 48 or 28
    local row = Instance.new("Frame")
    row.BackgroundTransparency = 1
    row.Size = UDim2.new(1, 0, 0, rowH)
    row.Parent = container

    local label = Instance.new("TextLabel")
    label.BackgroundTransparency = 1
    label.Size = UDim2.new(1, -(self.knob_w + 16), 1, 0)
    label.Font = Enum.Font.Gotham
    label.Text = name
    label.TextColor3 = Color3.fromRGB(220, 220, 230)
    label.TextSize = self.mobile and 15 or 13
    label.TextXAlignment = Enum.TextXAlignment.Left
    label.Parent = row

    local btn = Instance.new("TextButton")
    btn.BackgroundColor3 = default and Color3.fromRGB(0, 170, 255) or Color3.fromRGB(45, 45, 52)
    btn.BorderSizePixel = 0
    btn.Size = UDim2.fromOffset(self.knob_w, self.knob_h)
    btn.Position = UDim2.new(1, -self.knob_w, 0.5, -self.knob_h / 2)
    btn.Text = ""
    btn.AutoButtonColor = false
    btn.Parent = row

    local buc = Instance.new("UICorner")
    buc.CornerRadius = UDim.new(1, 0)
    buc.Parent = btn

    local knobD = self.knob_h - 6
    local knob = Instance.new("Frame")
    knob.BackgroundColor3 = Color3.fromRGB(240, 240, 245)
    knob.BorderSizePixel = 0
    knob.Size = UDim2.fromOffset(knobD, knobD)
    knob.Position = default and UDim2.new(1, -knobD - 3, 0.5, -knobD / 2) or UDim2.fromOffset(3, 3)
    knob.Parent = btn
    local kuc = Instance.new("UICorner")
    kuc.CornerRadius = UDim.new(1, 0)
    kuc.Parent = knob

    local state = default and true or false
    local function paint()
        btn.BackgroundColor3 = state and Color3.fromRGB(0, 170, 255) or Color3.fromRGB(45, 45, 52)
        knob.Position = state and UDim2.new(1, -knobD - 3, 0.5, -knobD / 2) or UDim2.fromOffset(3, 3)
    end

    btn.MouseButton1Click:Connect(function()
        state = not state
        paint()
        if cb then
            local ok, err = pcall(cb, state)
            if not ok then warn("[SHITARO UI] toggle cb:", err) end
        end
    end)

    return { value = function() return state end, set = function(v) state = v and true or false; paint() end }
end

function UI:addSlider(tab, section, name, default, min, max, round, cb, suffix)
    local container = section or tab.page.Frame
    local rowH = self.mobile and 68 or 42
    local row = Instance.new("Frame")
    row.BackgroundTransparency = 1
    row.Size = UDim2.new(1, 0, 0, rowH)
    row.Parent = container

    local label = Instance.new("TextLabel")
    label.BackgroundTransparency = 1
    label.Size = UDim2.new(1, -70, 0, 20)
    label.Font = Enum.Font.Gotham
    label.Text = name
    label.TextColor3 = Color3.fromRGB(220, 220, 230)
    label.TextSize = self.mobile and 15 or 13
    label.TextXAlignment = Enum.TextXAlignment.Left
    label.Parent = row

    local val = Instance.new("TextLabel")
    val.BackgroundTransparency = 1
    val.Size = UDim2.fromOffset(70, 20)
    val.Position = UDim2.new(1, -70, 0, 0)
    val.Font = Enum.Font.GothamBold
    val.Text = tostring(default) .. (suffix or "")
    val.TextColor3 = Color3.fromRGB(0, 170, 255)
    val.TextSize = self.mobile and 14 or 12
    val.TextXAlignment = Enum.TextXAlignment.Right
    val.Parent = row

    local barY = self.mobile and 34 or 26
    local barH = self.mobile and 14 or 6

    local bar = Instance.new("Frame")
    bar.BackgroundColor3 = Color3.fromRGB(40, 40, 48)
    bar.BorderSizePixel = 0
    bar.Size = UDim2.new(1, 0, 0, barH)
    bar.Position = UDim2.new(0, 0, 0, barY)
    bar.Parent = row

    local buc = Instance.new("UICorner")
    buc.CornerRadius = UDim.new(1, 0)
    buc.Parent = bar

    local fill = Instance.new("Frame")
    fill.BackgroundColor3 = Color3.fromRGB(0, 170, 255)
    fill.BorderSizePixel = 0
    fill.Size = UDim2.fromScale((default - min) / (max - min), 1)
    fill.Parent = bar
    local fuc = Instance.new("UICorner")
    fuc.CornerRadius = UDim.new(1, 0)
    fuc.Parent = fill

    local dragging = false
    local function setFromX(x)
        local abs = bar.AbsolutePosition.X
        local size = bar.AbsoluteSize.X
        if size <= 0 then return end
        local p = math.clamp((x - abs) / size, 0, 1)
        local raw = min + (max - min) * p
        if round then
            local mult = 10 ^ round
            raw = math.floor(raw * mult + 0.5) / mult
        else
            raw = math.floor(raw + 0.5)
        end
        fill.Size = UDim2.fromScale((raw - min) / (max - min), 1)
        val.Text = tostring(raw) .. (suffix or "")
        if cb then
            local ok, err = pcall(cb, raw)
            if not ok then warn("[SHITARO UI] slider cb:", err) end
        end
    end

    -- bigger hit area for touch: use row InputBegan
    local hitArea = Instance.new("TextButton")
    hitArea.BackgroundTransparency = 1
    hitArea.Text = ""
    hitArea.Size = UDim2.new(1, 0, 0, barH + (self.mobile and 24 or 8))
    hitArea.Position = UDim2.new(0, 0, 0, barY - (self.mobile and 12 or 4))
    hitArea.Parent = row

    hitArea.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1
            or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            setFromX(input.Position.X)
        end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement
            or input.UserInputType == Enum.UserInputType.Touch) then
            setFromX(input.Position.X)
        end
    end)
    UserInputService.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1
            or input.UserInputType == Enum.UserInputType.Touch then
            dragging = false
        end
    end)
end

function UI:addColor(tab, section, name, default, cb)
    local container = section or tab.page.Frame
    local rowH = self.mobile and 48 or 28
    local row = Instance.new("Frame")
    row.BackgroundTransparency = 1
    row.Size = UDim2.new(1, 0, 0, rowH)
    row.Parent = container

    local label = Instance.new("TextLabel")
    label.BackgroundTransparency = 1
    label.Size = UDim2.new(1, -70, 1, 0)
    label.Font = Enum.Font.Gotham
    label.Text = name
    label.TextColor3 = Color3.fromRGB(220, 220, 230)
    label.TextSize = self.mobile and 15 or 13
    label.TextXAlignment = Enum.TextXAlignment.Left
    label.Parent = row

    local swatchW = self.mobile and 56 or 34
    local swatchH = self.mobile and 34 or 18

    local swatch = Instance.new("TextButton")
    swatch.BackgroundColor3 = default
    swatch.BorderSizePixel = 0
    swatch.Size = UDim2.fromOffset(swatchW, swatchH)
    swatch.Position = UDim2.new(1, -swatchW, 0.5, -swatchH / 2)
    swatch.Text = ""
    swatch.AutoButtonColor = false
    swatch.Parent = row

    local suc = Instance.new("UICorner")
    suc.CornerRadius = UDim.new(0, 6)
    suc.Parent = swatch

    local h, s, v = Color3.toHSV(default)
    local function paint()
        swatch.BackgroundColor3 = Color3.fromHSV(h, s, v)
        if cb then pcall(cb, Color3.fromHSV(h, s, v)) end
    end

    -- left tap = shift hue, right tap = cycle sat
    swatch.MouseButton1Click:Connect(function()
        h = (h + 0.06) % 1
        paint()
    end)
    swatch.MouseButton2Click:Connect(function()
        s = s > 0.6 and 0.4 or 1
        paint()
    end)

    return {
        set = function(c)
            h, s, v = Color3.toHSV(c)
            paint()
        end
    }
end
