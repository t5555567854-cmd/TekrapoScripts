local roleCacheMod=nil
local roleCache={}
local roleCacheStamp=0
local function refreshRoleCache()
    local now=os.clock()
    if now-roleCacheStamp<0.4 and next(roleCache) then return end
    roleCacheStamp=now
    if not roleCacheMod then
        local ok,m=pcall(function() return require(Replicated:WaitForChild("Modules"):WaitForChild("CurrentRoundClient")) end)
        if ok and type(m)=="table" then roleCacheMod=m end
    end
    local d=roleCacheMod and roleCacheMod.PlayerData
    if type(d)~="table" then return end
    local new={}
    for name,info in pairs(d) do
        if type(info)=="table" and info.Role then new[name]=info.Role end
    end
    roleCache=new
end

local function alive(p)
    local c=p.Character; if not c then return false end
    local h=c:FindFirstChildOfClass("Humanoid")
    return h and h.Health>0
end
local function getRole(pl)
    local bp=pl:FindFirstChildOfClass("Backpack")
    local ch=pl.Character
    local function chk(t)
        if not t then return nil end
        if t:FindFirstChild("Knife") then return "m" end
        if t:FindFirstChild("Gun") then return "s" end
        return nil
    end
    return chk(bp) or chk(ch)
end
local function cachedRole(pl)
    refreshRoleCache()
    local c=roleCache[pl.Name]
    if c=="Murderer" then return "m" end
    if c=="Sheriff" or c=="Hero" then return "s" end
    return getRole(pl)
end

local function refresh_target()
    refreshRoleCache()
    local best,bd=nil,math.huge
    for _,p in ipairs(Players:GetPlayers()) do
        if p~=LP and alive(p) and cachedRole(p)=="m" then
            local h=p.Character and p.Character:FindFirstChild("Head")
            if h then
                local d=(h.Position-Cam.CFrame.Position).Magnitude
                if SA.antiJump then
                    local hv=p.Character:FindFirstChildOfClass("Humanoid")
                    if hv and hv.FloorMaterial==Enum.Material.Air then d = d * 0.5 end
                end
                if d<bd then bd=d; best=p end
            end
        end
    end
    if best~=target_p then target_p=best; target_char,target_part,target_hum=nil,nil,nil end
    if not best then return end
    local c=best.Character
    if c~=target_char then target_char=c; target_part,target_hum=nil,nil end
    if not c then return end
    if not target_part or not target_part.Parent then
        target_part = c:FindFirstChild("HumanoidRootPart") or c:FindFirstChild("UpperTorso") or c:FindFirstChild("Torso")
    end
    if not target_hum or not target_hum.Parent then
        target_hum = c:FindFirstChildOfClass("Humanoid")
    end
end

local function SA_getVel(root)
    local list = SA_hist[root]
    if not list or #list < 2 then return root.AssemblyLinearVelocity end
    local samples = {}
    local N = #list
    for i=2,N do
        local a, b = list[i-1], list[i]
        local dt = b[1] - a[1]
        if dt > 0.005 and dt < 0.3 then
            samples[#samples+1] = {
                vx = (b[2].X - a[2].X) / dt,
                vy = (b[2].Y - a[2].Y) / dt,
                vz = (b[2].Z - a[2].Z) / dt,
                w  = (i / N) * (1 / dt),
            }
        end
    end
    if #samples == 0 then return root.AssemblyLinearVelocity end
    local vx, vy, vz, w = 0, 0, 0, 0
    for _, s in ipairs(samples) do
        vx = vx + s.vx * s.w
        vy = vy + s.vy * s.w
        vz = vz + s.vz * s.w
        w = w + s.w
    end
    if w <= 0 then return root.AssemblyLinearVelocity end
    local v = Vector3.new(vx/w, vy/w, vz/w)
    if v.Magnitude > 200 then v = v.Unit * 200 end
    return v
end

local function SA_rayBlocked(a, b, ignore)
    local list = { workspace.CurrentCamera }
    for _, x in ipairs(ignore or {}) do table.insert(list, x) end
    local p = RaycastParams.new()
    p.FilterType = Enum.RaycastFilterType.Exclude
    for _ = 1, 8 do
        p.FilterDescendantsInstances = list
        local hit = workspace:Raycast(a, b - a, p)
        if not hit then return false end
        local i = hit.Instance
        if i.Transparency >= 0.9 or not i.CanCollide then
            table.insert(list, i)
        else
            return true
        end
    end
    return true
end

local SA_BODY = { "UpperTorso", "Torso", "LowerTorso", "Head", "HumanoidRootPart", "RightUpperArm", "LeftUpperArm", "Right Arm", "Left Arm" }
local function SA_anyVisible(targetChar, origin)
    for _, n in ipairs(SA_BODY) do
        local pt = targetChar:FindFirstChild(n)
        if pt and pt:IsA("BasePart") then
            if not SA_rayBlocked(origin, pt.Position, { targetChar }) then return true end
        end
    end
    return false
end

local function SA_findWbSpot(mRoot, muzzle)
    local mChar = mRoot.Parent
    local toM = muzzle - mRoot.Position
    local base = math.atan2(toM.Z, toM.X)
    for _, dist in ipairs({ 1.5, 2.5, 3.5, 5, 7, 9 }) do
        for k = 0, 23 do
            local ang = base + (k % 2 == 0 and 1 or -1) * math.ceil(k / 2) * (math.pi / 12)
            for _, h in ipairs({ 1, 2, 3, 4.5, 6 }) do
                local p = mRoot.Position + Vector3.new(math.cos(ang) * dist, h, math.sin(ang) * dist)
                if not SA_rayBlocked(p, mRoot.Position, { mChar }) and SA_anyVisible(mChar, p) then
                    return p
                end
            end
        end
    end
end

local function SA_findQuietSpot(mRoot, myChar)
    local root = myChar:FindFirstChild("HumanoidRootPart")
    if not root then return end
    local facing = CFrame.lookAt(root.Position, Vector3.new(mRoot.Position.X, root.Position.Y, mRoot.Position.Z))
    for _, off in ipairs({
        Vector3.new(0, 0, 4), Vector3.new(0, 0, -4),
        Vector3.new(4, 0, 0), Vector3.new(-4, 0, 0),
        Vector3.new(0, 5, 3), Vector3.new(0, 2, 2),
    }) do
        local pos = (facing * CFrame.new(off)).Position
        if not SA_rayBlocked(root.Position, pos) and not SA_rayBlocked(pos, mRoot.Position) then
            return CFrame.lookAt(pos, mRoot.Position)
        end
    end
end

local function resolve()
    if not SA.enabled then return nil end
    if not target_p or not target_p.Character then
        pcall(refresh_target)
        if not target_p then return nil end
    end
    local c = target_char
    if not c or not c.Parent then pcall(refresh_target); c = target_char end
    if not c then return nil end
    local root = c:FindFirstChild("HumanoidRootPart")
    if not root then return nil end

    local vel = SA_getVel(root)
    local hum2 = c:FindFirstChildOfClass("Humanoid")
    local lead = math.clamp(EC.rtt, 0.05, 0.45)
    if lead < 0.09 then lead = 0.09 end

    local moveDir = hum2 and hum2.MoveDirection or Vector3.zero
    local strafe = 0
    if moveDir.Magnitude > 0.1 then
        local right = root.CFrame.RightVector
        strafe = math.abs(right:Dot(moveDir.Unit))
    end
    lead = lead + strafe * 0.04

    local dy = 0
    if hum2 and hum2.FloorMaterial == Enum.Material.Air then
        local vy = root.AssemblyLinearVelocity.Y
        dy = vy * lead - 0.5 * workspace.Gravity * lead * lead
    else
        dy = math.clamp(vel.Y * lead, -3, 3)
    end
    if hum2 and hum2:GetState() == Enum.HumanoidStateType.Jumping then
        dy = dy + (hum2.JumpPower or 50) * 0.012
    end

    local off = Vector3.new(vel.X * lead, dy, vel.Z * lead)

    local myChar = LP.Character
    local myRoot = myChar and myChar:FindFirstChild("HumanoidRootPart")
    local att = myRoot and myRoot:FindFirstChild("GunRaycastAttachment")
    local muzzle = att and att.WorldPosition or (myRoot and myRoot.Position)
    if not muzzle then return nil end

    -- СТАРЫЙ ПРОСТРЕЛ: wbSpot как fromPos для расчёта, attachment НЕ двигаем
    local wbSpot
    if SA.wb and SA_rayBlocked(muzzle, root.Position, { c, myChar }) and not SA_anyVisible(c, muzzle) then
        wbSpot = SA_findWbSpot(root, muzzle)
    end

    local quietCf
    if SA.quiet and not wbSpot then
        quietCf = SA_findQuietSpot(root, myChar)
    end

    local order = SA.part == "Head"
        and { "Head", "UpperTorso", "Torso", "HumanoidRootPart", "LowerTorso" }
        or { "UpperTorso", "Torso", "LowerTorso", "HumanoidRootPart", "Head" }

    local fromPos = wbSpot or (quietCf and quietCf.Position) or muzzle
    rp.FilterDescendantsInstances = { myChar, workspace.CurrentCamera }

    local bestPt, bestDist, fallback
    for _, name in ipairs(order) do
        local part = c:FindFirstChild(name)
        if part and part:IsA("BasePart") then
            local aimPos = part.Position + off
            fallback = fallback or aimPos
            local dir = aimPos - fromPos
            local hit = workspace:Raycast(fromPos, dir, rp)
            if not hit or hit.Instance:IsDescendantOf(c) then
                local d = (aimPos - fromPos).Magnitude
                if not bestDist or d < bestDist then
                    bestDist = d; bestPt = aimPos
                end
            end
        end
    end
    if not bestPt and wbSpot then
        for _, name in ipairs(order) do
            local part = c:FindFirstChild(name)
            if part and part:IsA("BasePart") then
                local aimPos = part.Position + off
                local dir = aimPos - wbSpot
                local hit = workspace:Raycast(wbSpot, dir, rp)
                if not hit or hit.Instance:IsDescendantOf(c) then
                    return aimPos
                end
            end
        end
    end
    return bestPt or fallback or (root.Position + off)
end

install_hooks=function()
    local cur = get_ws_fn()
    if not cur then return end
    if cur ~= WSH.m then
        WSH.m = cur
        WSH.hooked = {}
    end
    if not WSH.hooked then WSH.hooked = {} end
    pcall(function() setreadonly(cur,false) end)

    local aimCandidates = {
        "GetMouseTargetCFrame","GetTargetPosition","GetTargetCFrame","GetAimPosition",
        "GetShootCFrame","GetShootTarget","GetBulletOrigin","CalculateAim",
        "ComputeTarget","GetFiringVector","GetShotOrigin","GetGunFireData",
    }
    for _, name in ipairs(aimCandidates) do
        if type(cur[name])=="function" and not WSH.hooked[name] then
            local orig = cur[name]
            cur[name] = function(self, ...)
                local pt = resolve()
                if pt then
                    local n = select("#", ...)
                    if n >= 2 then return pt end
                    return CFrame.new(pt)
                end
                return orig(self, ...)
            end
            WSH.hooked[name] = orig
        end
    end
end

ka_tick=function()
    if not KA.on then return end
    local c=LP.Character; if not c then return end
    local knife=c:FindFirstChild("Knife")
    if not knife then
        local bp=LP:FindFirstChildOfClass("Backpack")
        knife=bp and bp:FindFirstChild("Knife")
        if knife then
            local h=c:FindFirstChildOfClass("Humanoid")
            if h then pcall(function() h:EquipTool(knife) end) end
        end
        return
    end
    local now=os.clock()
    if now-KA.last<0.05 then return end
    local myp=c:FindFirstChild("HumanoidRootPart"); if not myp then return end
    local events=knife:FindFirstChild("Events")
    local stabbed=events and events:FindFirstChild("KnifeStabbed")
    local touched=events and events:FindFirstChild("HandleTouched")
    if not stabbed or not touched then return end
    local fired=false
    for _,plr in ipairs(Players:GetPlayers()) do
        if plr~=LP and alive(plr) then
            local tchar=plr.Character
            local tpart=tchar and (tchar:FindFirstChild("HumanoidRootPart") or tchar:FindFirstChild("Head"))
            if tpart and (tpart.Position-myp.Position).Magnitude<=KA.dist then
                if not fired then pcall(function() stabbed:FireServer() end); fired=true end
                pcall(function() touched:FireServer(tpart) end)
            end
        end
    end
    if fired then KA.last=now end
end

local function to_pos(v)
    if typeof(v)=="Vector3" then return v end
    if typeof(v)=="CFrame" then return v.Position end
    if typeof(v)=="Instance" then
        if v:IsA("Attachment") then return v.WorldPosition end
        if v:IsA("BasePart") then return v.Position end
    end
end
enableTracer_fn=function()
    if TRACE.conn then return end
    local m=get_ws_fn()
    local ev=m and m.GunFired
    if typeof(ev)~="Instance" then return end
    TRACE.conn=ev.OnClientEvent:Connect(function(gun,a,b)
        if not fl.tracer then return end
        local c=LP.Character; if not c then return end
        if typeof(gun)~="Instance" or not gun:IsDescendantOf(c) then return end
        local pa,pb=to_pos(a),to_pos(b)
        if not (pa and pb) then
            local cam=workspace.CurrentCamera; if not cam then return end
            pa=pa or cam.CFrame.Position
            pb=pb or (pa+cam.CFrame.LookVector*3000)
        end
        local d=pb-pa; local dist=d.Magnitude
        if dist<0.1 then return end
        local p=Instance.new("Part")
        p.Name="sBeam"; p.Anchored=true
        p.CanCollide=false; p.CanQuery=false; p.CanTouch=false; p.CastShadow=false
        p.Material=Enum.Material.Neon; p.Color=FC.tracer
        p.Size=Vector3.new(SZ.tracer,SZ.tracer,dist)
        p.CFrame=CFrame.lookAt(pa+d/2,pb)
        p.Transparency=0.05; p.Parent=workspace
        local li=Instance.new("PointLight",p)
        li.Color=FC.tracer; li.Brightness=3; li.Range=10
        local ts=0
        local cn
        cn=RS.Heartbeat:Connect(function(dt)
            ts=ts+dt; local k=ts/0.6
            if k>=1 or not p.Parent then cn:Disconnect(); if p.Parent then p:Destroy() end; return end
            p.Transparency=0.05+k*0.95
            li.Brightness=3*(1-k)
        end)
    end)
end
disableTracer_fn=function()
    if TRACE.conn then pcall(function() TRACE.conn:Disconnect() end); TRACE.conn=nil end
end
connect_trace_fn=enableTracer_fn

bt_build=function()
    bt_destroy()
    local char=LP.Character; if not char then return end
    local rhrp=char:FindFirstChild("HumanoidRootPart"); if not rhrp then return end
    char.Archivable=true
    local ok,m=pcall(function() return char:Clone() end)
    char.Archivable=false
    if not ok or not m then return end
    local rparts={}
    for _,o in char:GetDescendants() do if o:IsA("BasePart") then rparts[#rparts+1]=o end end
    local ci=0
    for _,o in m:GetDescendants() do
        if o:IsA("Script") or o:IsA("LocalScript") or o:IsA("ModuleScript")
        or o:IsA("Decal") or o:IsA("Texture") or o:IsA("ParticleEmitter") or o:IsA("Beam")
        or o:IsA("Trail") or o:IsA("Light") or o:IsA("Sound") then
            pcall(function() o:Destroy() end)
        elseif o:IsA("BasePart") then
            o.Anchored=true; o.CanCollide=false; o.CanQuery=false
            o.CanTouch=false; o.Massless=true; o.CastShadow=false
            if o.Name=="HumanoidRootPart" then o.Transparency=1
            else o.Material=Enum.Material.ForceField; o.Color=Color3.fromRGB(170,100,255); o.Transparency=0.3 end
            ci=ci+1
            BT.cache[#BT.cache+1]={o,rparts[ci]}
        end
    end
    local hum2=m:FindFirstChildOfClass("Humanoid")
    if hum2 then pcall(function() hum2:Destroy() end) end
    m.Name="\0"; m.Parent=workspace
    BT.model=m
end
bt_update=function()
    if not BT.on then return end
    local char=LP.Character
    local rhrp=char and char:FindFirstChild("HumanoidRootPart"); if not rhrp then return end
    if not BT.model or not BT.model.Parent then bt_build(); if not BT.model then return end end
    local now=os.clock()
    local base_cf=rhrp.CFrame
    if BT.n<BT.cap then BT.n=BT.n+1 else BT.first=BT.first%BT.cap+1 end
    local idx=(BT.first+BT.n-2)%BT.cap+1
    local slot=BT.hist[idx]; if not slot then slot={}; BT.hist[idx]=slot end
    slot[1]=now; slot[2]=base_cf
    if now-BT.lastping>=0.2 then
        BT.lastping=now
        local ok,val=pcall(function() return Stats.Network.ServerStatsItem["Data Ping"]:GetValue()/1000 end)
        BT.ping=math.clamp((ok and val) or 0.15,0.05,0.6)
    end
    local target=now-BT.ping
    local cf=base_cf
    for k=BT.n,1,-1 do
        local s=BT.hist[(BT.first+k-2)%BT.cap+1]
        if s and s[1] and s[1]<=target then cf=s[2]; break end
    end
    while BT.n>0 and BT.hist[BT.first] and BT.hist[BT.first][1]<now-1 do
        BT.first=BT.first%BT.cap+1; BT.n=BT.n-1
    end
    local baseInv=rhrp.CFrame:Inverse()
    for i=1,#BT.cache do
        local cp,rpv=BT.cache[i][1],BT.cache[i][2]
        if cp and cp.Parent and rpv and rpv.Parent then
            cp.CFrame=cf*(baseInv*rpv.CFrame)
        end
    end
end
bt_conn=RS.Heartbeat:Connect(function() if BT.on then pcall(bt_update) end end)

local function flingCore(thrp,thum,time)
    local c=LP.Character
    local hrp=c and c:FindFirstChild("HumanoidRootPart"); if not hrp then return false end
    local hum2=c and c:FindFirstChildOfClass("Humanoid"); if not hum2 then return false end
    if thum and thum.Sit then return false end
    local orig=hrp.CFrame
    local cam=workspace.CurrentCamera
    local oldSubj=cam.CameraSubject
    local oldFall=workspace.FallenPartsDestroyHeight
    pcall(function() workspace.FallenPartsDestroyHeight=0/0 end)
    pcall(function() cam.CameraSubject=thrp end)
    local bv=Instance.new("BodyVelocity")
    bv.MaxForce=Vector3.new(9e9,9e9,9e9) bv.Velocity=Vector3.zero bv.Parent=hrp
    local se=hum2:GetStateEnabled(Enum.HumanoidStateType.Seated)
    pcall(function() hum2:SetStateEnabled(Enum.HumanoidStateType.Seated,false) end)
    local t0=os.clock()
    local dur=time or 1.6
    local ang=0
    while os.clock()-t0<dur and hrp.Parent and thrp.Parent do
        local tv=thrp.Velocity
        if tv.Magnitude>50 and thum then tv=thum.MoveDirection*thum.WalkSpeed end
        local lead=thrp.Position+tv*0.08
        ang=ang+165
        local r=ang%360
        local off
        if r<90 then off=Vector3.new(0,2.5,0)
        elseif r<180 then off=Vector3.new(0,-2.5,0)
        elseif r<270 then off=Vector3.new(2.5,0,0)
        else off=Vector3.new(-2.5,0,0) end
        local cf=CFrame.new(lead+off)*CFrame.Angles(math.rad(ang),0,0)
        pcall(function()
            hrp.CFrame=cf
            c:SetPrimaryPartCFrame(cf)
            hrp.Velocity=Vector3.new(9e8,9e8*10,9e8)
            hrp.RotVelocity=Vector3.new(9e9,9e9,9e9)
        end)
        RS.Heartbeat:Wait()
    end
    if bv.Parent then bv:Destroy() end
    pcall(function() hum2:SetStateEnabled(Enum.HumanoidStateType.Seated,se) end)
    pcall(function() cam.CameraSubject=hum2 end)
    if hrp.Parent then
        pcall(function()
            hrp.CFrame=orig
            c:SetPrimaryPartCFrame(orig)
            hum2:ChangeState(Enum.HumanoidStateType.GettingUp)
            hrp.Velocity=Vector3.zero
        end)
    end
    pcall(function() workspace.FallenPartsDestroyHeight=oldFall end)
    if oldSubj then pcall(function() cam.CameraSubject=oldSubj end) end
    return true
end
local function flingTarget(target)
    if TROLL.flinging then return end
    local tchar=target and target.Character; if not tchar then return end
    local thrp=tchar:FindFirstChild("HumanoidRootPart") or tchar:FindFirstChild("Head"); if not thrp then return end
    local thum=tchar:FindFirstChildOfClass("Humanoid")
    TROLL.flinging=true
    task.spawn(function() flingCore(thrp,thum,2.2); TROLL.flinging=false end)
end
local function flingNearest() local t=nearest_player(); if t then flingTarget(t) end end

                    end
                    h.FillTransparency=SZ.esp; h.FillColor=col; h.OutlineColor=col
                elseif fl.espGhost and roleCache[pl.Name]=="Murderer" then

        l.TextStrokeTransparency=0; l.TextStrokeColor3=Color3.new(0,0,0)

    espTxtTags[pl]=nil

local function updateRoleTags()
    for _,pl in ipairs(Players:GetPlayers()) do
        if pl~=LP then
            local c=pl.Character
            local h=c and c:FindFirstChild("Head")
            local r=c and roleLabel(pl) or nil
            if h and r then attachRoleTag(pl,h,r) else detachRoleTag(pl) end
        else detachRoleTag(pl) end
    end
end

local function mkJump()
    if jumpC then jumpC:Disconnect() end
    local c=LP.Character; if not c then return end
    local h=c:FindFirstChildOfClass("Humanoid"); if not h then return end
    jumpC=h.Jumping:Connect(function(active)
        if not active then return end
        local hrp=HRP(); if not hrp then return end
        local pos=hrp.Position-Vector3.new(0,2.5,0)
        local col=FC.jump
        local function ring()
            local p=Instance.new("Part")
            p.Size=Vector3.new(1,1,1); p.Anchored=true
            p.CanCollide=false; p.CanQuery=false; p.CanTouch=false; p.CastShadow=false
            p.Material=Enum.Material.Neon; p.Color=col; p.Transparency=0.1
            p.CFrame=CFrame.new(pos+Vector3.new(0,0.7,0))*CFrame.Angles(math.rad(90),0,0)
            local m=Instance.new("SpecialMesh",p)
            m.MeshId="rbxassetid://3270017"
            m.Scale=Vector3.new(SZ.jump,SZ.jump,0.3)
            p.Parent=workspace
            local ts=0
            local cn
            cn=RS.Heartbeat:Connect(function(dt)
                ts=ts+dt; local k=ts/0.5
                if k>=1 or not p.Parent then cn:Disconnect(); if p.Parent then p:Destroy() end; return end
                local sz=SZ.jump+k*6
                m.Scale=Vector3.new(sz,sz,0.3*(1-k*0.5))
                p.Transparency=0.1+k*0.9
            end)
        end
        if jStyle=="Classic" then ring()
        elseif jStyle=="Double" then ring(); task.delay(0.15,ring)
        elseif jStyle=="Ripple" then for i=1,3 do task.delay(i*0.08,ring) end
        elseif jStyle=="Particles" then
            ring()
            local a=Instance.new("Attachment",workspace.Terrain)
            a.WorldPosition=pos
            local pe=Instance.new("ParticleEmitter",a)
            pe.Texture="rbxasset://textures/particles/sparkles_main.dds"
            pe.Rate=0; pe.Lifetime=NumberRange.new(0.6,1)
            pe.Speed=NumberRange.new(6,10); pe.SpreadAngle=Vector2.new(180,180)
            pe.Size=NumberSequence.new({NumberSequenceKeypoint.new(0,0.8),NumberSequenceKeypoint.new(1,0)})
            pe.Transparency=NumberSequence.new({NumberSequenceKeypoint.new(0,0.1),NumberSequenceKeypoint.new(1,1)})
            pe.Color=ColorSequence.new(col,LT(col))
            pe.LightEmission=1; pe.LightInfluence=0; pe.Drag=3
            pe:Emit(35)
            task.delay(1.2,function() if a.Parent then a:Destroy() end end)
        else ring() end
    end)
end
local function clJump() if jumpC then jumpC:Disconnect(); jumpC=nil end end

local function killAll()
    local c=LP.Character; if not c then return end
    local knife=c:FindFirstChild("Knife")
    if not knife then
        local bp=LP:FindFirstChildOfClass("Backpack")
        knife=bp and bp:FindFirstChild("Knife")
        if knife then
            local h=c:FindFirstChildOfClass("Humanoid")
            if h then pcall(function() h:EquipTool(knife) end) end
            task.wait(0.3); knife=c:FindFirstChild("Knife")
        end
    end
    if not knife then return end
    local events=knife:FindFirstChild("Events")
    local stabbed=events and events:FindFirstChild("KnifeStabbed")
    local touched=events and events:FindFirstChild("HandleTouched")
    if not stabbed or not touched then return end
    local myp=c:FindFirstChild("HumanoidRootPart"); if not myp then return end
    local orig=myp.CFrame
    for _,plr in ipairs(Players:GetPlayers()) do
        if plr~=LP and alive(plr) then
            local tchar=plr.Character
            local tpart=tchar and (tchar:FindFirstChild("HumanoidRootPart") or tchar:FindFirstChild("Head"))
            if tpart then
                pcall(function() myp.CFrame=tpart.CFrame+Vector3.new(0,0,2) end)
                task.wait(0.1)
                pcall(function() stabbed:FireServer() end)
                pcall(function() touched:FireServer(tpart) end)
                task.wait(0.05)
            end
        end
    end
    pcall(function() myp.CFrame=orig end)
end

-- ============================================================
--  MAIN LOOP
-- ============================================================
task.spawn(function()
    while ADDON.loaded do
        pcall(function()
            local now = os.clock()
            if now - (ADDON.role_acc or 0) >= 0.05 then
                ADDON.role_acc = now
                refresh_target()
            end
            sample_ping()
            track_step(now)
            ka_tick()
            if fl.roleTag and now - (ADDON.rtag_acc or 0) >= 0.25 then
                ADDON.rtag_acc = now
                pcall(updateRoleTags)
            end
            if fl.antifling and not MOVE.fly then
                local ac = LP.Character
                local ahrp = ac and ac:FindFirstChild("HumanoidRootPart")
                if ahrp then
                    pcall(function()
                        if ahrp.AssemblyLinearVelocity.Magnitude > 60 then
                            ahrp.AssemblyLinearVelocity = Vector3.zero
                            ahrp.Velocity = Vector3.zero
                        end
                        ahrp.AssemblyAngularVelocity = Vector3.zero
                        ahrp.RotVelocity = Vector3.zero
                    end)
                end
            end
            if now - (ADDON.hook_acc or 0) >= 0.5 then
                ADDON.hook_acc = now
                install_hooks()
                if TRACE.on then connect_trace_fn() end
            end
        end)
        task.wait()
    end
end)

task.spawn(function()
    while ADDON.loaded do pcall(hookKills); task.wait(0.5) end
end)

-- ============================================================
--  UNLOAD
-- ============================================================
ADDON.unload = function()
    ADDON.loaded = false
    SA.enabled=false; KS.enabled=false; KA.on=false
    TRACE.on=false; autograb_on=false; BT.on=false
    grab_desc_stop()
    getgenv().KNIFE_AIM_RESOLVE = nil
    moveReset(); stopTroll(); stopAntiJump()
    workspace.Gravity = 196.2
    if TRACE.conn then pcall(function() TRACE.conn:Disconnect() end); TRACE.conn=nil end
    if ADDON.charConn then pcall(function() ADDON.charConn:Disconnect() end); ADDON.charConn=nil end
    if bt_conn then pcall(function() bt_conn:Disconnect() end); bt_conn=nil end
    pcall(bt_destroy)
    for _, fn in ipairs({clTrail,clParts,clSky,clSnow,clBo,clHat,clLight,clOrb,clGhost,clFoot,clRB,clGlow,clJump,clAtmo,clFB,clSkybox,clBloom,clESP,disableTracer_fn}) do
        pcall(fn)
    end
    for pl in pairs(espTxtTags) do detachRoleTag(pl) end
    pcall(function() if fl._fog then toggleFog(false) end end)
    local m = WSH.m
    if m and WSH.hooked then
        pcall(function() setreadonly(m,false) end)
        for name, orig in pairs(WSH.hooked) do
            pcall(function() m[name] = orig end)
        end
        WSH.hooked = nil; WSH.m = nil
    end
    local core = game.CoreGui:FindFirstChild("Kavo")
    if core then pcall(function() core:Destroy() end) end
    getgenv().UN_ADDON = nil
    print("[addon] unloaded")
end
