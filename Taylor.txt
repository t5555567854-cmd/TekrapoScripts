-- ============================================================
-- gui attach cascade
-- ============================================================
local function attach_gui(gui)
    if not gui then return nil end
    local candidates = {}
    if type(gethui) == "function" then
        local ok, h = pcall(gethui)
        if ok and h then candidates[#candidates+1] = h end
    end
    candidates[#candidates+1] = CoreGuiSvc
    local pg = LP:FindFirstChildOfClass("PlayerGui")
    if pg then candidates[#candidates+1] = pg end

    for i = 1, #candidates do
        local parent = candidates[i]
        local ok = pcall(function() gui.Parent = parent end)
        if ok and gui.Parent then return parent end
    end
    return nil
end

local UI = {}

function UI.new(title)
    local self = setmetatable({}, UI)
    self.title = title or "SHITARO"
    self.enabled = true
    self.key = Enum.KeyCode.RightShift
    self.tabs = {}
    self.tab = nil
    self.mobile = UserInputService.TouchEnabled and true or false

    if self.mobile then
        self.size = Vector2.new(340, 480)
        self.knob_w = 58
        self.knob_h = 32
    else
        self.size = Vector2.new(520, 400)
        self.knob_w = 40
        self.knob_h = 20
    end

    local vp = CAM and CAM.ViewportSize or Vector2.new(800, 600)
    if self.mobile then
        self.pos = Vector2.new(math.max(10, (vp.X - self.size.X) / 2), 40)
    else
        self.pos = Vector2.new(60, 60)
    end

    self.ok, self.err = pcall(function() self:_build() end)
    if not self.ok then
        warn("[SHITARO] UI build failed:", self.err)
    end
    return self
end

function UI:toggle(state)
    if not self.win then return end
    self.enabled = state and true or not self.enabled
    self.win.Visible = self.enabled
end

    btn.Size = UDim2.new(1, 0, 0, btnH)
    btn.Font = Enum.Font.GothamBold
    btn.Text = name
    btn.TextColor3 = Color3.fromRGB(200, 200, 210)
    btn.TextSize = self.mobile and 15 or 13
    btn.TextXAlignment = Enum.TextXAlignment.Center
    btn.Parent = self.tab_strip

    local buc = Instance.new("UICorner")

    buc.Parent = btn

    tab.btn = btn
    self.tabs[#self.tabs + 1] = tab

    btn.MouseButton1Click:Connect(function()
        self:selectTab(tab)
    end)

    if #self.tabs == 1 then
        self:selectTab(tab)
    end
    return tab

function UI:selectTab(tab)
    self.tab = tab
    for _, t in ipairs(self.tabs) do
        if t == tab then
            t.btn.BackgroundColor3 = Color3.fromRGB(0, 170, 255)
            t.btn.TextColor3 = Color3.fromRGB(255, 255, 255)
        else
            t.btn.BackgroundColor3 = Color3.fromRGB(24, 24, 28)
            t.btn.TextColor3 = Color3.fromRGB(160, 160, 170)
        end
    end
    if not tab.page then
        tab.page = self:_buildPage(tab)
    end
    for _, t in ipairs(self.tabs) do
        if t.page then t.page.Visible = (t == tab) end
    end
end

function UI:addSlider(tab, section, name, default, min, max, round, cb, suffix)
    if not self.ok or not tab or not tab.page then return end
    local container = section or tab.page.Frame
    local rowH = self.mobile and 68 or 42

    local row = Instance.new("Frame")
    row.BackgroundTransparency = 1
    row.Size = UDim2.new(1, 0, 0, rowH)
    row.Parent = container

    local label = Instance.new("TextLabel")
    label.BackgroundTransparency = 1
    label.Size = UDim2.new(1, -70, 0, 20)
    label.Font = Enum.Font.Gotham
    label.Text = name
    label.TextColor3 = Color3.fromRGB(220, 220, 230)
    label.TextSize = self.mobile and 15 or 13
    label.TextXAlignment = Enum.TextXAlignment.Left
    label.Parent = row

    local val = Instance.new("TextLabel")
    val.BackgroundTransparency = 1
    val.Size = UDim2.fromOffset(70, 20)
    val.Position = UDim2.new(1, -70, 0, 0)
    val.Font = Enum.Font.GothamBold
    val.Text = tostring(default) .. (suffix or "")
    val.TextColor3 = Color3.fromRGB(0, 170, 255)
    val.TextSize = self.mobile and 14 or 12
    val.TextXAlignment = Enum.TextXAlignment.Right
    val.Parent = row

    local barY = self.mobile and 34 or 26
    local barH = self.mobile and 14 or 6

    local bar = Instance.new("Frame")
    bar.BackgroundColor3 = Color3.fromRGB(40, 40, 48)
    bar.BorderSizePixel = 0
    bar.Size = UDim2.new(1, 0, 0, barH)
    bar.Position = UDim2.new(0, 0, 0, barY)
    bar.Parent = row

    local buc = Instance.new("UICorner")
    buc.CornerRadius = UDim.new(1, 0)
    buc.Parent = bar

    local fill = Instance.new("Frame")
    fill.BackgroundColor3 = Color3.fromRGB(0, 170, 255)
    fill.BorderSizePixel = 0
    fill.Size = UDim2.fromScale((default - min) / (max - min), 1)
    fill.Parent = bar

    local fuc = Instance.new("UICorner")
    fuc.CornerRadius = UDim.new(1, 0)
    fuc.Parent = fill

    local dragging = false
    local function setFromX(x)
        local abs = bar.AbsolutePosition.X
        local size = bar.AbsoluteSize.X
        if size <= 0 then return end
        local p = math.clamp((x - abs) / size, 0, 1)
        local raw = min + (max - min) * p
        if round then
            local mult = 10 ^ round
            raw = math.floor(raw * mult + 0.5) / mult
        else
            raw = math.floor(raw + 0.5)
        end
        fill.Size = UDim2.fromScale((raw - min) / (max - min), 1)
        val.Text = tostring(raw) .. (suffix or "")
        if cb then
            local ok, err = pcall(cb, raw)
            if not ok then warn("[SHITARO UI] slider cb:", err) end
        end
    end

    local hitArea = Instance.new("TextButton")
    hitArea.BackgroundTransparency = 1
    hitArea.Text = ""
    hitArea.Size = UDim2.new(1, 0, 0, barH + (self.mobile and 24 or 8))
    hitArea.Position = UDim2.new(0, 0, 0, barY - (self.mobile and 12 or 4))
    hitArea.Parent = row

    hitArea.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1
            or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            setFromX(input.Position.X)
        end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement
            or input.UserInputType == Enum.UserInputType.Touch) then
            setFromX(input.Position.X)
        end
    end)

local BT = { hist = {}, keep = 2.0 }

local function bt_track()
    local now = os.clock()
    local cutoff = now - BT.keep
    for _, plr in ipairs(Players:GetPlayers()) do
        if plr == LP then continue end
        local char = plr.Character
        local hrp = char and char:FindFirstChild("HumanoidRootPart")
        local hum = char and char:FindFirstChildOfClass("Humanoid")
        if hrp and hum and hum.Health > 0 then
            local h = BT.hist[plr.Name]
            if not h then h = {}; BT.hist[plr.Name] = h end
            h[#h + 1] = { t = now, pos = hrp.Position }
            local i = 1
            while i <= #h and h[i].t < cutoff do i = i + 1 end
            if i > 1 then
                for j = 1, #h - (i - 1) do h[j] = h[j + i - 1] end
                for j = #h - (i - 1) + 1, #h do h[j] = nil end
            end
        end
    end
    for name in pairs(BT.hist) do
        if not Players:FindFirstChild(name) then BT.hist[name] = nil end
    end
end

local function bt_get(name, ms_back)
    local h = BT.hist[name]
    if not h or #h == 0 then return nil end
    local target = os.clock() - (ms_back / 1000)
    local best, best_d = h[1], math.huge
    for i = 1, #h do
        local d = math.abs(h[i].t - target)
        if d < best_d then best, best_d = h[i], d end
    end
    return best and best.pos or nil
end

local target_player = nil
local target_char = nil
local target_part = nil
local target_hum = nil

local function holds(c, n) return c ~= nil and c:FindFirstChild(n) ~= nil end

local function refresh_target()
    local found = nil
    local m = get_round()
    local data = m and m.PlayerData
    if type(data) == "table" then
        for name, d in pairs(data) do
            if type(d) == "table" and d.Role == "Murderer" and not d.Dead then
                found = Players:FindFirstChild(name)
                if found then break end
            end
        end
    end
    if not found then
        for _, plr in ipairs(Players:GetPlayers()) do
            if plr ~= LP and holds(plr.Character, "Knife") then
                found = plr
                break
            end
        end
    end
    if not found then
        local mychar = LP.Character
        local myhrp = mychar and mychar:FindFirstChild("HumanoidRootPart")
        if myhrp then
            local best, bestd = nil, math.huge
            for _, plr in ipairs(Players:GetPlayers()) do
                if plr ~= LP and plr.Character then
                    local h = plr.Character:FindFirstChild("HumanoidRootPart")
                    local hum = plr.Character:FindFirstChildOfClass("Humanoid")
                    if h and hum and hum.Health > 0 then
                        local d = (h.Position - myhrp.Position).Magnitude
                        if d < bestd then best, bestd = plr, d end
                    end
                end
            end
            found = best
        end
    end

    if found ~= target_player then
        target_player = found
        target_char = nil
        target_part = nil
        target_hum = nil
    end
    if not found then return end
    local char = found.Character
    if char ~= target_char then
        target_char = char
        target_part = nil
        target_hum = nil
    end
    if not char then return end
    if not target_part or not target_part.Parent then
        target_part = char:FindFirstChild("HumanoidRootPart")
            or char:FindFirstChild("UpperTorso")
            or char:FindFirstChild("Torso")
    end
    if not target_hum or not target_hum.Parent then
        target_hum = char:FindFirstChildOfClass("Humanoid")
    end
end

local function target_alive()
    if not target_part or not target_part.Parent then return false end
    if not target_hum or not target_hum.Parent then return false end
    return target_hum.Health > 0
end

-- ============================================================
-- motion tracking
-- ============================================================
local SNAP = { n = 48, i = 0, count = 0, t = {}, p = {}, last_pos = nil, last_time = 0, vel = Vector3.zero }
for i = 1, SNAP.n do SNAP.t[i] = 0; SNAP.p[i] = Vector3.zero end

local function snap_push(now, pos)
    SNAP.i = SNAP.i % SNAP.n + 1
    SNAP.t[SNAP.i] = now
    SNAP.p[SNAP.i] = pos
    if SNAP.count < SNAP.n then SNAP.count = SNAP.count + 1 end
end

local function snap_get(k)
    local idx = (SNAP.i - k - 1) % SNAP.n + 1
    return SNAP.t[idx], SNAP.p[idx]
end

local function fit_velocity()
    if SNAP.count < 3 then return nil end
    local newest = snap_get(0)
    local used, sum_d = 0, 0
    for k = 0, SNAP.count - 1 do
        local t = snap_get(k)
        if newest - t > 0.4 then break end
        used = used + 1
        sum_d = sum_d + (t - newest)
    end
    if used < 3 then return nil end
    local mean_d = sum_d / used
    local num = Vector3.zero
    local den = 0
    for k = 0, used - 1 do
        local t, p = snap_get(k)
        local d = (t - newest) - mean_d
        num = num + p * d
        den = den + d * d
    end
    if den < 1e-8 then return nil end
    return num / den
end

local function recent_velocity()
    if SNAP.count < 2 then return nil end
    local newest, head = snap_get(0)
    local best = nil
    for k = 1, SNAP.count - 1 do
        local t, p = snap_get(k)
        local dt = newest - t
        if dt > 0.35 then break end
        best = (head - p) / dt
    end
    return best
end

local function track_motion(now)
    if not target_part or not target_part.Parent then
        SNAP.count = 0
        SNAP.i = 0
        SNAP.last_pos = nil
        return
    end
    local pos = target_part.Position
    if not SNAP.last_pos then
        SNAP.last_pos = pos
        SNAP.last_time = now
        snap_push(now, pos)
        return
    end
    if now - SNAP.last_time < 0.005 then return end
    SNAP.last_time = now
    SNAP.last_pos = pos
    snap_push(now, pos)

    local fit = fit_velocity()
    local fast = recent_velocity()
    if fit and fast then
        local sf = Vector3.new(fit.X, 0, fit.Z).Magnitude
        local ss = Vector3.new(fast.X, 0, fast.Z).Magnitude
        if sf > 200 or sf ~= sf then fit = nil end
        if ss > 200 or ss ~= ss then fast = nil end
    end
    if fit and fast then
        SNAP.vel = Vector3.new(fast.X * 0.7 + fit.X * 0.3, fast.Y, fast.Z * 0.7 + fit.Z * 0.3)
    elseif fast then
        SNAP.vel = fast
    elseif fit then
        SNAP.vel = fit
    end
    local h = Vector3.new(SNAP.vel.X, 0, SNAP.vel.Z)
    if h.Magnitude > 120 then
        SNAP.vel = Vector3.new(SNAP.vel.X * 0.4, SNAP.vel.Y, SNAP.vel.Z * 0.4)
    end
end

local function clear_motion()
    SNAP.count = 0; SNAP.i = 0
    SNAP.last_pos = nil; SNAP.last_time = 0
    SNAP.vel = Vector3.zero
end

-- ping
-- ============================================================
local PING = { seen = false, rtt = 0 }
local function sample_ping()
    local ok, ms = pcall(function()
        return Stats.Network.ServerStatsItem["Data Ping"]:GetValue()
    end)
    if ok and type(ms) == "number" and ms == ms and ms > 0 then
        local s = math.clamp(ms / 1000, 0, 1)
        if PING.seen then PING.rtt = PING.rtt * 0.85 + s * 0.15
        else PING.rtt = s; PING.seen = true end
    end
end

local function lead_time()
    if not PING.seen then return 0 end
    return math.clamp(PING.rtt, 0, 0.5)
end

-- aim
-- ============================================================
local HIT_NAMES = {
    "HumanoidRootPart", "UpperTorso", "Torso", "LowerTorso", "Head",
    "RightUpperArm", "LeftUpperArm", "RightUpperLeg", "LeftUpperLeg",
}

local function aim_point()
    if not target_char or not target_char.Parent then return nil end
    if not target_hum or target_hum.Health <= 0 then return nil end

    if CONFIG.backtrack then
        local btpos = bt_get(target_player and target_player.Name or "", CONFIG.bt_window)
        if btpos then return btpos end
    end

    local part = target_part
    if not part then
        for _, n in ipairs(HIT_NAMES) do
            local p = target_char:FindFirstChild(n)
            if p then part = p; break end
        end
    end
    if not part then return nil end

    local base = part.Position
    if not CONFIG.predict then return base end
    local t = lead_time()
    if t <= 0 then return base end
    local v = SNAP.vel
    local x = base.X + v.X * t
    local z = base.Z + v.Z * t
    local y = base.Y
    if math.abs(v.Y) > 1 then
        y = base.Y + v.Y * t - 0.5 * workspace.Gravity * t * t
    end
    return Vector3.new(x, y, z)
end

local last_shot = 0
local function auto_fire(now)
    if not CONFIG.auto or not CONFIG.aim or not target_alive() then return end
    if my_role() ~= "Sheriff" then return end
    local gun, eq = get_gun()
    if not gun then return end
    if not eq then
        local hum = LP.Character and LP.Character:FindFirstChildOfClass("Humanoid")
        if hum then pcall(function() hum:EquipTool(gun) end) end
        return
    end
    local delay = math.max(CONFIG.auto_delay, 0.05)
    if now - last_shot < delay then return end
    local p = aim_point()
    if not p then return end
    local remote = gun:FindFirstChild("Shoot")
    if remote and remote:IsA("RemoteEvent") then
        local c = LP.Character
        local hrp = c and c:FindFirstChild("HumanoidRootPart")
        local att = hrp and hrp:FindFirstChild("GunRaycastAttachment")
        local oc = att and att.WorldCFrame or (hrp and hrp.CFrame)
        if oc then
            pcall(function() remote:FireServer(oc, CFrame.new(p)) end)
            last_shot = now
        end
    end
end

                if self_chams_backup[d] == nil then
                    self_chams_backup[d] = { d.Material, d.Color, d.Transparency }

                elseif mode == "Glass" then
                    d.Material = Enum.Material.Glass
                    d.Transparency = 0.5
                    d.Color = CONFIG.self_cham_col
                end
            end
        end
    end
end

                        e.box.Size = Vector2.new(w, h)
                        e.box.Visible = true
                    else
                        e.box.Visible = false
                    end

                    if CONFIG.esp_name then
                        e.name.Color = CONFIG.esp_name_col
                        e.name.Text = plr.DisplayName
                        e.name.Position = Vector2.new(bot.X, boxY - 16)
                        e.name.Visible = true
                    else
                        e.name.Visible = false
                    end

                        local myhrp = LP.Character and LP.Character:FindFirstChild("HumanoidRootPart")
                        local d = myhrp and (hrp.Position - myhrp.Position).Magnitude or 0
                        e.dist.Color = CONFIG.esp_dist_col
                        e.dist.Text = string.format("[%d]", math.floor(d + 0.5))
                        e.dist.Position = Vector2.new(bot.X, bot.Y + 2)
                        e.dist.Visible = true
                    else
                        e.dist.Visible = false
                    end
                else
                    e.box.Visible = false
                    e.name.Visible = false
                    e.dist.Visible = false
                end
            end
        else
            if e then
                e.box.Visible = false
                e.name.Visible = false
                e.dist.Visible = false
            end
        end
    end
end

getgenv().SHITARO_UNLOAD = function()
    pcall(function() if hb then hb:Disconnect() end end)
    pcall(function() if render_conn then render_conn:Disconnect() end end)
    pcall(function() if tracer_conn then tracer_conn:Disconnect() end end)
    pcall(self_chams_restore)
    pcall(function() if self_esp_hl then self_esp_hl:Destroy() end end)
    pcall(function() if self_hl then self_hl:Destroy() end end)
    for plr, e in pairs(esp_cache) do
        for _, d in pairs(e) do pcall(function() d:Remove() end) end
    end
    table.clear(esp_cache)
    pcall(function() if fov_circle then fov_circle:Remove() end end)
    pcall(function() if MENU.frame then MENU.frame:Destroy() end end)
    notification("SHITARO unloaded.")
end
